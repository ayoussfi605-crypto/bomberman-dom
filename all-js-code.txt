import { test, mock } from "node:test";
import assert from "node:assert/strict";
import { State } from "../public/src/framework/state/index.js";

test("default notification stays synchronous and preserves unrelated state", () => {
  const store = new State({ count: 0, label: "keep" });
  const seen = [];
  store.subscribe((state, version) => seen.push([state, version]));
  store.setState({ count: 1 });
  assert.deepEqual(seen, [[{ count: 1, label: "keep" }, 1]]);
});

test("microtask batching merges immediately and notifies once with latest version", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const seen = [];
  store.subscribe((state, version) => seen.push([state.count, version]));
  store.setState({ count: 1 });
  store.setState(state => ({ count: state.count + 1 }));
  store.setState(state => ({ count: state.count + 1 }));
  assert.equal(store.getState().count, 3);
  assert.deepEqual(seen, []);
  await Promise.resolve();
  assert.deepEqual(seen, [[3, 3]]);
  store.setState({ count: 4 });
  await Promise.resolve();
  assert.deepEqual(seen, [[3, 3], [4, 4]]);
});

test("invalid/no-op patches preserve state, version and notification count", () => {
  const store = new State({ count: NaN, data: [] });
  const before = store.getState();
  const listener = mock.fn();
  store.subscribe(listener);
  for (const patch of [null, undefined, false, 1, "text", [], {}, { count: NaN }, { data: before.data }]) store.setState(patch);
  assert.equal(store.getState(), before);
  assert.equal(store.version, 0);
  assert.equal(listener.mock.callCount(), 0);
});

test("listener errors do not prevent other subscribers from running", t => {
  const log = t.mock.method(console, "error", () => {});
  const store = new State({ count: 0 });
  const listener = mock.fn();
  store.subscribe(() => { throw new Error("test listener"); });
  store.subscribe(listener);
  store.setState({ count: 1 });
  assert.equal(log.mock.callCount(), 1);
  assert.equal(listener.mock.callCount(), 1);
});

test("unsubscribe before scheduled notification removes the subscriber", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const listener = mock.fn();
  const off = store.subscribe(listener);
  store.setState({ count: 1 });
  off(); off();
  await Promise.resolve();
  assert.equal(listener.mock.callCount(), 0);
});

test("self-unsubscribe does not skip the next listener", () => {
  const store = new State({ count: 0 });
  const seen = [];
  const off = store.subscribe(() => { seen.push("first"); off(); });
  store.subscribe(() => seen.push("second"));
  store.setState({ count: 1 });
  store.setState({ count: 2 });
  assert.deepEqual(seen, ["first", "second", "second"]);
});

test("updates from microtask listeners schedule another notification", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const seen = [];
  store.subscribe(state => {
    seen.push(state.count);
    if (state.count === 1) store.setState({ count: 2 });
  });
  store.setState({ count: 1 });
  await Promise.resolve();
  await Promise.resolve();
  assert.deepEqual(seen, [1, 2]);
});

test("raf batches and schedules again after notification", t => {
  const callbacks = [];
  const previous = Object.getOwnPropertyDescriptor(globalThis, "requestAnimationFrame");
  Object.defineProperty(globalThis, "requestAnimationFrame", {
    configurable: true, writable: true, value: callback => callbacks.push(callback),
  });
  t.after(() => {
    if (previous) Object.defineProperty(globalThis, "requestAnimationFrame", previous);
    else delete globalThis.requestAnimationFrame;
  });
  const store = new State({ count: 0 }, { batch: "raf" });
  const listener = mock.fn();
  store.subscribe(listener);
  store.setState({ count: 1 }); store.setState({ count: 2 });
  assert.equal(callbacks.length, 1);
  assert.equal(listener.mock.callCount(), 0);
  callbacks.shift()();
  assert.equal(listener.mock.callCount(), 1);
  assert.equal(listener.mock.calls[0].arguments[0].count, 2);
  store.setState({ count: 3 });
  assert.equal(callbacks.length, 1);
});

test("invalid batch modes and subscriber types fail clearly", () => {
  assert.throws(() => new State({}, { batch: "later" }), TypeError);
  assert.throws(() => new State().subscribe(null), TypeError);
});

  if (server.listening) await new Promise(resolve => server.close(resolve));
  await rm(profile, { recursive: true, force: true });

import {
  ERR,
  LOBBY_COUNTDOWN_MS,
  LOBBY_WAIT_MS,
  MAX_PLAYERS,
  MIN_PLAYERS,
  NICK_MAX,
  NICK_MIN,
  NICK_REGEX,
} from "../../shared/constants.js";

/** @returns {string | null} an ERR code, or null when the nickname is usable. */
export function validateNickname(raw, takenNicknames = []) {
  if (typeof raw !== "string") return ERR.NICK_INVALID;

  const nickname = raw.trim();
  if (nickname.length < NICK_MIN || nickname.length > NICK_MAX)
    return ERR.NICK_INVALID;
  if (!NICK_REGEX.test(nickname)) return ERR.NICK_INVALID;

  const lower = nickname.toLowerCase();
  if (takenNicknames.some((other) => other.toLowerCase() === lower))
    return ERR.NICK_TAKEN;

  return null;
}

export function createLobby({
  now = () => Date.now(),
  maxPlayers = MAX_PLAYERS,
  waitMs = LOBBY_WAIT_MS,
  countdownMs = LOBBY_COUNTDOWN_MS,
} = {}) {
  const players = new Map();
  let timer = null;

  function nicknames() {
    return [...players.values()].map((player) => player.nickname);
  }

  function startTimer(kind, durationMs) {
    timer = { kind, endsAt: now() + durationMs };
  }

  function join(id, rawNickname) {
    if (players.has(id)) return { ok: false, code: ERR.ALREADY_JOINED };

    const code = validateNickname(rawNickname, nicknames());
    if (code) return { ok: false, code };
    if (players.size >= maxPlayers) return { ok: false, code: ERR.ROOM_FULL };

    const nickname = rawNickname.trim();
    players.set(id, { id, nickname });

    if (players.size >= maxPlayers) {
      if (!timer || timer.kind !== "start") startTimer("start", countdownMs);
    } else if (players.size >= MIN_PLAYERS && !timer) {
      startTimer("wait", waitMs);
    }

    return { ok: true, nickname };
  }

  function leave(id) {
    const removed = players.delete(id);
    if (removed && players.size < MIN_PLAYERS) timer = null;
    return removed;
  }

  /** Call every ~100 ms. @returns null | { event: "start", players: [{id, nickname}] } */
  function update() {
    if (!timer) return null;
    if (now() < timer.endsAt) return null;

    if (timer.kind === "wait") {
      startTimer("start", countdownMs);
      return null;
    }

    const entries = [...players.values()].map((player) => ({
      id: player.id,
      nickname: player.nickname,
    }));
    timer = null;

    if (entries.length < MIN_PLAYERS) return null; // safety net
    return { event: "start", players: entries };
  }

  function view() {
    const list = nicknames();
    let timerView = null;

    if (timer) {
      const remaining = Math.max(0, timer.endsAt - now());
      timerView = {
        kind: timer.kind,
        secondsLeft: Math.ceil(remaining / 1000),
      };
    }

    return {
      players: list,
      count: list.length,
      max: maxPlayers,
      timer: timerView,
    };
  }

  return { join, leave, update, view, has: (id) => players.has(id) };
}

import {
  C2S,
  COLS,
  ERR,
  LOBBY_COUNTDOWN_MS,
  LOBBY_WAIT_MS,
  MAX_PLAYERS,
  ROWS,
  S2C,
  TICK_MS,
} from "../../shared/constants.js";
import { createChatGate, sanitizeText } from "../chat.js";
import { errorMessage } from "../net/validate.js";
import { createLobby } from "./lobby.js";
import { createLoop } from "./loop.js";

export function createRoom(options = {}) {
  const {
    createGame = null,
    now = () => Date.now(),
    maxPlayers = MAX_PLAYERS,
    waitMs = LOBBY_WAIT_MS,
    countdownMs = LOBBY_COUNTDOWN_MS,
    tickMs = TICK_MS,
    lobbyPollMs = 100,
    logger = console,
  } = options;

  const sessions = new Set();
  const sessionsById = new Map();
  const chatGate = createChatGate({ now });

  let lobby = createLobby({ now, maxPlayers, waitMs, countdownMs });
  let phase = "lobby";
  let game = null;
  let loop = null;
  let pollTimer = null;
  let lastLobbyJson = null;
  let stopped = false;

  let tickCount = 0;
  let lastRateLog = now();

  function broadcast(type, data) {
    const payload = JSON.stringify({ type, data });
    for (const session of sessions) {
      if (!session.joined) continue;
      if (typeof session.sendRaw === "function") session.sendRaw(payload);
      else session.send(type, data);
    }
  }

  function broadcastChat(from, text) {
    broadcast(S2C.CHAT, { from, text, ts: now() });
  }

  function fail(session, code) {
    session.send(S2C.ERROR, { code, message: errorMessage(code) });
  }

  function currentCount() {
    if (phase === "playing") {
      let count = 0;
      for (const session of sessions) {
        if (session.slot !== null && session.slot !== undefined) count += 1;
      }
      return count;
    }
    return lobby.view().count;
  }

  function helloData() {
    return { phase, count: currentCount(), max: maxPlayers };
  }

  function stopPoll() {
    if (pollTimer !== null) {
      clearInterval(pollTimer);
      pollTimer = null;
    }
  }

  function ensurePoll() {
    if (stopped || pollTimer !== null || phase !== "lobby") return;
    pollTimer = setInterval(tickLobby, lobbyPollMs);
    if (typeof pollTimer.unref === "function") pollTimer.unref();
  }

  function broadcastLobbyIfChanged() {
    const view = lobby.view();
    const json = JSON.stringify(view);
    if (json === lastLobbyJson) return;
    lastLobbyJson = json;
    broadcast(S2C.LOBBY, view);
  }

  function tickLobby() {
    if (stopped || phase !== "lobby") return;

    const started = lobby.update();
    if (started && started.event === "start") {
      startGame(started.players);
      return;
    }
    broadcastLobbyIfChanged();
  }

  function startGame(entries) {
    if (typeof createGame !== "function") {
      logger.error?.(
        "[room] no game engine: server/game/game.js does not exist yet, the match cannot start",
      );
      return;
    }

    const players = entries.map((entry, slot) => ({
      slot,
      nickname: entry.nickname,
    }));

    let nextGame;
    try {
      nextGame = createGame(
        players.map((player) => ({ ...player })),
        now(),
      );
    } catch (error) {
      logger.error?.("[room] could not create the game:", error);
      return;
    }

    game = nextGame;
    entries.forEach((entry, slot) => {
      const session = sessionsById.get(entry.id);
      if (session) session.slot = slot;
    });

    phase = "playing";
    stopPoll();
    lastLobbyJson = null;

    const tiles = game.getTiles();
    const receivers = [];
    for (const entry of entries) {
      const session = sessionsById.get(entry.id);
      if (session && session.joined) receivers.push(session);
    }

    for (const session of receivers) {
      session.send(S2C.START, {
        you: session.slot,
        cols: COLS,
        rows: ROWS,
        players,
        tiles,
      });
    }

    broadcast(S2C.SNAPSHOT, game.getSnapshot());

    loop = createLoop(stepGame, tickMs);
    loop.start();
  }

  function logTickRate() {
    tickCount += 1;
    const at = now();
    if (at - lastRateLog < 1000) return;
    const rate = (tickCount * 1000) / (at - lastRateLog);
    logger.log?.(`[room] snapshot rate: ${rate.toFixed(1)} Hz`);
    tickCount = 0;
    lastRateLog = at;
  }

  function stepGame(dtMs) {
    if (!game) return;

    const result = game.step(dtMs);

    if (result && result.tilesChanged)
      broadcast(S2C.TILES, { tiles: game.getTiles() });

    broadcast(S2C.SNAPSHOT, game.getSnapshot());
    logTickRate();

    const outcome = game.getResult();
    if (outcome) endGame(outcome);
  }

  function endGame(result) {
    broadcast(S2C.GAMEOVER, result);

    if (loop) {
      loop.stop();
      loop = null;
    }
    game = null;

    for (const session of sessions) {
      session.joined = false;
      session.slot = null;
      session.nickname = null;
    }
    lobby = createLobby({ now, maxPlayers, waitMs, countdownMs });
    lastLobbyJson = null;
    phase = "lobby";

    for (const session of sessions) session.send(S2C.HELLO, helloData());
    ensurePoll();
  }

  function handleJoin(session, data) {
    if (phase === "playing") return fail(session, ERR.GAME_IN_PROGRESS);
    if (session.joined) return fail(session, ERR.ALREADY_JOINED);

    const result = lobby.join(session.id, data.nickname);
    if (!result.ok) return fail(session, result.code);

    session.joined = true;
    session.nickname = result.nickname;
    session.slot = null;

    session.send(S2C.JOINED, { nickname: result.nickname });
    broadcastLobbyIfChanged();
    broadcastChat(null, `${result.nickname} joined`);
    tickLobby();
  }

  function handleChat(session, data) {
    if (!session.joined) return fail(session, ERR.NOT_JOINED);
    if (!chatGate.allow(session.id)) return fail(session, ERR.RATE_LIMITED);

    const text = sanitizeText(data.text);
    if (!text) return;

    broadcastChat(session.nickname, text);
  }

  function handleGameMessage(session, type, data) {
    if (!session.joined || phase !== "playing" || !game) return;
    if (session.slot === null || session.slot === undefined) return;

    if (type === C2S.INPUT) game.setInput(session.slot, data.dir);
    else if (type === C2S.BOMB) game.placeBomb(session.slot);
  }

  function handle(session, type, data) {
    switch (type) {
      case C2S.JOIN:
        return handleJoin(session, data);
      case C2S.CHAT:
        return handleChat(session, data);
      case C2S.INPUT:
      case C2S.BOMB:
        return handleGameMessage(session, type, data);
      default:
        return fail(session, ERR.UNKNOWN_TYPE);
    }
  }

  function handleConnect(session) {
    if (stopped) return;
    sessions.add(session);
    sessionsById.set(session.id, session);
    session.send(S2C.HELLO, helloData());
    ensurePoll();
  }

  function handleDisconnect(session) {
    if (!sessions.has(session)) return;
    sessions.delete(session);
    sessionsById.delete(session.id);
    chatGate.forget(session.id);

    const wasJoined = session.joined;
    const nickname = session.nickname;
    const slot = session.slot;
    session.joined = false;
    session.slot = null;

    if (phase === "playing" && game) {
      if (slot !== null && slot !== undefined) game.removePlayer(slot);
      if (wasJoined) broadcastChat(null, `${nickname} left`);
      return;
    }

    if (!wasJoined) return;
    lobby.leave(session.id);
    broadcastChat(null, `${nickname} left`);
    tickLobby();
  }

  function stop() {
    stopped = true;
    stopPoll();
    if (loop) {
      loop.stop();
      loop = null;
    }
    game = null;
  }

  return {
    handle,
    handleConnect,
    handleDisconnect,
    stop,
    get phase() {
      return phase;
    },
    get playerCount() {
      return currentCount();
    },
  };
}

  } = {},
) {
  const tickNs = BigInt(Math.round(tickMs * 1_000_000));

  let running = false;
  let handle = null;
  let last = 0n;
  let carry = 0n;

  function pump() {
    if (!running) return;

    const current = now();
    const elapsed = current > last ? current - last : 0n;
    last = current;
    carry += elapsed;

    let steps = 0;
    while (carry >= tickNs && steps < maxCatchUp) {
      carry -= tickNs;
      steps += 1;
    }
    if (carry >= tickNs) carry %= tickNs;

    for (let i = 0; i < steps; i += 1) stepFn(tickMs);

    handle = schedule(pump, SCHEDULE_MS);
    if (handle && typeof handle.unref === "function") handle.unref();
  }

  function start() {
    if (running) return;
    running = true;
    last = now();
    carry = 0n;
    pump();
  }

  function stop() {
    running = false;
    if (handle !== null) {
      cancel(handle);
      handle = null;
    }
  }

  return {
    start,
    stop,
    get running() {
      return running;
    },
  };
}

import { ERR, S2C } from "../../shared/constants.js";
import { createSession } from "./session.js";
import { errorMessage, parseMessage } from "./validate.js";

const MAX_BAD_MESSAGES = 10;
const WS_OPEN = 1;
const WS_CONNECTING = 0;

function closeSocket(ws, code, reason) {
  try {
    if (ws.readyState === WS_OPEN || ws.readyState === WS_CONNECTING)
      ws.close(code, reason);
  } catch {
    // Already gone.
  }
}

export function attachTransport({ wss, room, logger = console, now } = {}) {
  function badMessage(session, code) {
    session.badMessages += 1;
    session.send(S2C.ERROR, { code, message: errorMessage(code) });
    if (session.badMessages >= MAX_BAD_MESSAGES) {
      closeSocket(session.ws, 1008, "Too many bad messages");
    }
  }

  function handleMessage(session, raw) {
    if (session.closed) return;

    if (!session.limiter.take()) {
      session.send(S2C.ERROR, {
        code: ERR.RATE_LIMITED,
        message: errorMessage(ERR.RATE_LIMITED),
      });
      return;
    }

    const parsed = parseMessage(raw);
    if (!parsed.ok) {
      badMessage(session, parsed.code);
      return;
    }

    try {
      room.handle(session, parsed.type, parsed.data);
    } catch (error) {
      logger.error?.("[transport] handler failed:", error);
    }
  }

  function onConnection(ws, request) {
    try {
      request?.socket?.setNoDelay?.(true);
    } catch {
      // Optional tuning only.
    }

    const session = createSession(ws, { now });

    ws.on("error", (error) => {
      logger.error?.("[transport] socket error:", error?.message ?? error);
    });

    ws.on("message", (raw, isBinary) => {
      try {
        if (isBinary) {
          closeSocket(ws, 1003, "Text messages only");
          return;
        }
        handleMessage(session, raw);
      } catch (error) {
        logger.error?.("[transport] message failed:", error);
      }
    });

    ws.on("close", () => {
      if (session.closed) return;
      session.closed = true;
      try {
        room.handleDisconnect(session);
      } catch (error) {
        logger.error?.("[transport] disconnect failed:", error);
      }
    });

    try {
      room.handleConnect(session);
    } catch (error) {
      logger.error?.("[transport] connect failed:", error);
    }
  }

  wss.on("connection", onConnection);
}

import { MAX_MESSAGES_PER_SECOND } from "../../shared/constants.js";

export function createRateLimiter({
  limit = MAX_MESSAGES_PER_SECOND,
  windowMs = 1000,
  now = () => Date.now(),
} = {}) {
  let hits = [];

  return {
    /** Records a hit when allowed. @returns {boolean} true if the event is allowed. */
    take() {
      const at = now();
      while (hits.length > 0 && at - hits[0] >= windowMs) hits.shift();
      if (hits.length >= limit) return false;
      hits.push(at);
      return true;
    },
    reset() {
      hits = [];
    },
    get count() {
      return hits.length;
    },
  };
}

import {
  C2S,
  CHAT_MAX_LEN,
  DIR,
  ERR,
  MAX_MESSAGE_BYTES,
} from "../../shared/constants.js";

const CLIENT_TYPES = new Set(Object.values(C2S));
const DIRECTIONS = new Set(Object.values(DIR));

const ERROR_MESSAGES = {
  [ERR.BAD_MESSAGE]: "Malformed message",
  [ERR.UNKNOWN_TYPE]: "Unknown message type",
  [ERR.NICK_INVALID]:
    "Nickname must be 2-12 characters (letters, digits, _ or -)",
  [ERR.NICK_TAKEN]: "Nickname already used",
  [ERR.ROOM_FULL]: "Room is full",
  [ERR.GAME_IN_PROGRESS]: "A game is already in progress",
  [ERR.ALREADY_JOINED]: "You have already joined",
  [ERR.NOT_JOINED]: "Join the room before sending this message",
  [ERR.RATE_LIMITED]: "Slow down",
};

export function errorMessage(code) {
  return ERROR_MESSAGES[code] ?? "Something went wrong";
}

function fail(code) {
  return { ok: false, code };
}

function isPlainObject(value) {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function toText(raw) {
  if (typeof raw === "string") return raw;
  if (raw instanceof Uint8Array) return Buffer.from(raw).toString("utf8");
  if (raw instanceof ArrayBuffer)
    return Buffer.from(new Uint8Array(raw)).toString("utf8");
  return null;
}

/**
 * @param {string | Uint8Array | ArrayBuffer} raw
 * @returns {{ ok: true, type: string, data: object } | { ok: false, code: string }}
 */
export function parseMessage(raw) {
  try {
    const text = toText(raw);
    if (text === null) return fail(ERR.BAD_MESSAGE);
    if (Buffer.byteLength(text, "utf8") > MAX_MESSAGE_BYTES)
      return fail(ERR.BAD_MESSAGE);

    let message;
    try {
      message = JSON.parse(text);
    } catch {
      return fail(ERR.BAD_MESSAGE);
    }

    if (!isPlainObject(message)) return fail(ERR.BAD_MESSAGE);

    const { type } = message;
    if (typeof type !== "string" || !CLIENT_TYPES.has(type))
      return fail(ERR.UNKNOWN_TYPE);

    const payload = message.data;

    switch (type) {
      case C2S.BOMB: {
        if (payload === undefined || payload === null)
          return { ok: true, type, data: {} };
        if (!isPlainObject(payload)) return fail(ERR.BAD_MESSAGE);
        return { ok: true, type, data: {} };
      }

      case C2S.JOIN: {
        if (!isPlainObject(payload) || typeof payload.nickname !== "string") {
          return fail(ERR.BAD_MESSAGE);
        }
        return { ok: true, type, data: { nickname: payload.nickname } };
      }

      case C2S.CHAT: {
        if (!isPlainObject(payload) || typeof payload.text !== "string") {
          return fail(ERR.BAD_MESSAGE);
        }
        if (payload.text.length > CHAT_MAX_LEN) return fail(ERR.BAD_MESSAGE);
        return { ok: true, type, data: { text: payload.text } };
      }

      case C2S.INPUT: {
        if (!isPlainObject(payload) || !("dir" in payload))
          return fail(ERR.BAD_MESSAGE);
        const { dir } = payload;
        if (dir !== null && !DIRECTIONS.has(dir)) return fail(ERR.BAD_MESSAGE);
        return { ok: true, type, data: { dir } };
      }

      default:
        return fail(ERR.UNKNOWN_TYPE);
    }
  } catch {
    return fail(ERR.BAD_MESSAGE);
  }
}

const CONTROL_CHARS = /[\u0000-\u001f\u007f-\u009f]/g;

/**
 * Trims, strips control characters and caps the length.
 * @returns {string | null} null when there is nothing to send.
 */
export function sanitizeText(raw) {
  if (typeof raw !== "string") return null;

  const text = raw.replace(CONTROL_CHARS, " ").replace(/\s+/g, " ").trim();
  if (!text) return null;

  return text.length > CHAT_MAX_LEN ? text.slice(0, CHAT_MAX_LEN) : text;
}

export function createChatGate({
  now = () => Date.now(),
  limit = CHAT_RATE_MAX,
  windowMs = CHAT_RATE_WINDOW_MS,
} = {}) {
  const limiters = new Map();

  return {
    allow(playerId) {
      let limiter = limiters.get(playerId);
      if (!limiter) {
        limiter = createRateLimiter({ limit, windowMs, now });
        limiters.set(playerId, limiter);
      }
      return limiter.take();
    },
    forget(playerId) {
      limiters.delete(playerId);
    },
    reset() {
      limiters.clear();
    },
  };
}

const MIME = {

function mimeType(filePath) {
  return (
    MIME[path.extname(filePath).toLowerCase()] ?? "application/octet-stream"
  );
}

function sendText(res, status, text, method) {
  res.writeHead(status, {
    "Content-Type": "text/plain; charset=utf-8",
    "Cache-Control": "no-store",
  });
  res.end(method === "HEAD" ? undefined : `${status} ${text}\n`);
}

/** Maps a request path to a file inside public/ or shared/, or returns a status. */
function resolveFile(rawPath) {
  let pathname;
  try {
    pathname = decodeURIComponent(rawPath);
  } catch {
    return { status: 400, text: "Bad Request" };
  }

  if (pathname.includes("\0")) return { status: 400, text: "Bad Request" };

  let root = PUBLIC_DIR;
  let relative = pathname;
  if (pathname === "/shared" || pathname.startsWith("/shared/")) {
    root = SHARED_DIR;
    relative = pathname.slice("/shared".length) || "/";
  }

  if (relative === "/") relative = "/index.html";

  // Resolve first, then verify the result is still inside the root.
  const filePath = path.resolve(root, `.${relative}`);
  if (filePath !== root && !filePath.startsWith(root + path.sep)) {
    return { status: 403, text: "Forbidden" };
  }

  return { filePath };
}

async function handleRequest(req, res) {
  const method = req.method ?? "GET";
  if (method !== "GET" && method !== "HEAD")
    return sendText(res, 405, "Method Not Allowed", method);

  const rawPath = (req.url ?? "/").split("?")[0].split("#")[0] || "/";
  const target = resolveFile(rawPath);
  if (target.status) return sendText(res, target.status, target.text, method);

  let info;
  try {
    info = await stat(target.filePath);
  } catch {
    return sendText(res, 404, "Not Found", method);
  }
  if (!info.isFile()) return sendText(res, 404, "Not Found", method);

  res.writeHead(200, {
    "Content-Type": mimeType(target.filePath),
    "Content-Length": info.size,
    "Cache-Control": "no-store",
  });

  if (method === "HEAD") {
    res.end();
    return;
  }

  const stream = createReadStream(target.filePath);
  stream.on("error", () => res.destroy());
  stream.pipe(res);
}

export function createAppServer({ logger = console, room: injectedRoom } = {}) {
  const server = http.createServer((req, res) => {
    handleRequest(req, res).catch((error) => {
      logger.error?.("[http] request failed:", error);
      if (!res.headersSent)
        sendText(res, 500, "Internal Server Error", req.method);
      else res.destroy();
    });
  });

  const wss = new WebSocketServer({
    server,
    perMessageDeflate: false,
    maxPayload: MAX_MESSAGE_BYTES,
  });

  const room = injectedRoom ?? createRoom({ logger });
  attachTransport({ wss, room, logger });

  // Heartbeat: ping every 30 s, terminate a socket that missed the previous ping.
  wss.on("connection", (ws) => {
    ws.isAlive = true;
    ws.on("pong", () => {
      ws.isAlive = true;
    });
  });

  const heartbeat = setInterval(() => {
    for (const ws of wss.clients) {
      if (ws.isAlive === false) {
        ws.terminate();
        continue;
      }
      ws.isAlive = false;
      try {
        ws.ping();
      } catch {
        ws.terminate();
      }
    }
  }, HEARTBEAT_MS);
  if (typeof heartbeat.unref === "function") heartbeat.unref();

  server.on("close", () => {
    clearInterval(heartbeat);
    room.stop?.();
    try {
      wss.close();
    } catch {
      // Already closing.
    }
  });

  server.on("error", (error) => logger.error?.("[http] server error:", error));
  wss.on("error", (error) => logger.error?.("[ws] server error:", error));

  return server;
}

// Framework Client-Side Router (Hash-based)

export function createRouter(routes = {}) {
  // Get current route path from URL hash (e.g. "#/active" -> "/active")
  function getPath() {
    const hash = window.location.hash.slice(1); // remove '#'
    return hash || '/'; // Default route if hash is empty
  }

  // Find matching route and run its callback function
  function handleRoute() {
    const currentPath = getPath();
    const routeCallback = routes[currentPath] || routes['/']; // Fallback to '/' if route not found
    
    if (typeof routeCallback === 'function') {
      routeCallback(currentPath);
    }
  }

  return {
    // Navigate programmatically to a new path (e.g., router.navigate('/active'))
    navigate(path) {
      window.location.hash = path;
    },

    // Start listening for URL hash changes and execute initial route
    start() {
      // Listen to browser hash changes (Back/Forward buttons or clicking links)
      window.addEventListener('hashchange', handleRoute);

      // Execute route callback immediately on page load
      handleRoute();
    }
  };
}

import createDom from "./createDom.js";
import applyStyle from "./style.js";

function updateProps(element, oldProps = {}, newProps = {}) {
  const keys = new Set([...Object.keys(oldProps), ...Object.keys(newProps)]);

  for (const key of keys) {
    const oldValue = oldProps[key];
    const newValue = newProps[key];

    if (oldValue === newValue) continue;
    if (key === "key") continue;

    if (key === "style") {
      applyStyle(element, oldValue, newValue);
      continue;
    }

    if (key.startsWith("on")) {
      addEvent(element, key, newValue);
      continue;
    }

    if (key === "value" && "value" in element) {
      const value = newValue == null ? "" : String(newValue);

      if (element.value !== value) {
        element.value = value;
      }

      continue;
    }

    if (key === "checked" && "checked" in element) {
      element.checked = Boolean(newValue);
      continue;
    }

    if (key === "selected" && "selected" in element) {
      element.selected = Boolean(newValue);
      continue;
    }
    if (newValue == null || newValue === false) {
      element.removeAttribute(key);
      continue;
    }

    if (newValue === true) {
      element.setAttribute(key, "");
      continue;
    }

    element.setAttribute(key, String(newValue));
  }
}

// Reconcile a list of children positionally inside `parent`.
function patchChildren(parent, oldChildren = [], newChildren = []) {
  const sharedLength = Math.min(oldChildren.length, newChildren.length);

  for (let i = 0; i < sharedLength; i++) {
    updateDOM(parent, newChildren[i], oldChildren[i], i);
  }

  for (let i = oldChildren.length - 1; i >= newChildren.length; i--) {
    const childToRemove = parent.childNodes[i];
    if (childToRemove) {
      removeEvents(childToRemove);
      parent.removeChild(childToRemove);
    }
  }

  for (let i = oldChildren.length; i < newChildren.length; i++) {
    parent.appendChild(createDom(newChildren[i]));
  }
}

function updateDOM(parent, newVNode, oldVNode, index = 0) {
  // Rendered vnodes are immutable. Reusing one skips the entire subtree.
  if (newVNode === oldVNode) return;

  const current = parent.childNodes[index];

  if (newVNode == null) {
    if (current) {
      removeEvents(current); 
      parent.removeChild(current);
    }
    return;
  }

  if (oldVNode == null) {
    parent.insertBefore(createDom(newVNode), current ?? null);
    return;
  }

  const newIsPrimitive =
    typeof newVNode === "string" || typeof newVNode === "number";

  const oldIsPrimitive =
    typeof oldVNode === "string" || typeof oldVNode === "number";

  if (newIsPrimitive || oldIsPrimitive) {
    if (
      newIsPrimitive &&
      oldIsPrimitive &&
      typeof newVNode === typeof oldVNode
    ) {
      if (String(newVNode) !== String(oldVNode)) {
        current.nodeValue = String(newVNode);
      }
    } else {
      removeEvents(current); 
      parent.replaceChild(createDom(newVNode), current);
    }

    return;
  }

  if (newVNode.type !== oldVNode.type) {
    removeEvents(current); 
    parent.replaceChild(createDom(newVNode), current);
    return;
  }

  updateProps(current, oldVNode.props, newVNode.props);

  patchChildren(current, oldVNode.children ?? [], newVNode.children ?? []);
}

export default function patch(oldVNode, newVNode, container) {
  if (oldVNode === newVNode) return;

  // A fragment root is a flat list of siblings inside the container.
  if (Array.isArray(oldVNode) || Array.isArray(newVNode)) {
    patchChildren(
      container,
      Array.isArray(oldVNode) ? oldVNode : [oldVNode],
      Array.isArray(newVNode) ? newVNode : [newVNode]
    );
    return;
  }

  updateDOM(container, newVNode, oldVNode, 0);
}

export default function createDom(vnode) {
  // A fragment (array of vnodes) renders as sibling nodes with no wrapper element.
  if (Array.isArray(vnode)) {
    const fragment = document.createDocumentFragment();

    vnode.forEach((child) => {
      fragment.appendChild(createDom(child));
    });

    return fragment;
  }

  if (typeof vnode === "string" || typeof vnode === "number") {
    return document.createTextNode(vnode);
  }

  let element = document.createElement(vnode.type);

  if (vnode.props) {
     addprop(vnode, element);
  }

  if (vnode.children) {
    vnode.children.forEach((child) => {
      element.appendChild(createDom(child));
    });
  }

  return element;
}

function clearStyle(element) {
  // Synchronize pending CSSOM writes before removing the attribute.
  if (element.hasAttribute("style")) element.removeAttribute("style");
}

function isStyleObject(value) {
  if (value === null || typeof value !== "object") return false;
  const prototype = Object.getPrototypeOf(value);
  return prototype === Object.prototype || prototype === null;
}

function setStyleProp(element, name, value) {
  const text = value == null || value === false ? "" : String(value);
  if (name.includes("-")) {
    if (text === "") element.style.removeProperty(name);
    else element.style.setProperty(name, text);
  } else {
    element.style[name] = text;
  }
}

export default function applyStyle(element, oldStyle, newStyle) {
  if (newStyle == null || newStyle === false || newStyle === "") {
    clearStyle(element);
    return;
  }
  if (typeof newStyle === "string") {
    element.setAttribute("style", newStyle);
    return;
  }
  if (!isStyleObject(newStyle)) {
    throw new TypeError("style must be a CSS string or a plain object");
  }
  if (!isStyleObject(oldStyle)) {
    clearStyle(element);
    for (const [name, value] of Object.entries(newStyle))
      setStyleProp(element, name, value);
    return;
  }
  for (const name of Object.keys(oldStyle)) {
    if (!Object.hasOwn(newStyle, name)) setStyleProp(element, name, "");
  }
  for (const [name, value] of Object.entries(newStyle)) {
    if (oldStyle[name] !== value) setStyleProp(element, name, value);
  }
}

const BATCH_MODES = ["sync", "microtask", "raf"];

export class State {
  constructor(initialState = {}, { batch = "sync" } = {}) {
    if (!BATCH_MODES.includes(batch)) {
      throw new TypeError(`State: batch must be one of ${BATCH_MODES.join(", ")}`);
    }
    this.state = initialState;
    this.listeners = [];
    this.version = 0;
    this.batch = batch;
    this.pending = false;
  }

  getState() {
    return this.state;
  }

  setState(update) {
    const changes = typeof update === "function" ? update(this.state) : update;
    if (changes == null || typeof changes !== "object" || Array.isArray(changes)) return;
    const keys = Object.keys(changes);
    if (keys.length === 0 || keys.every(key => Object.is(this.state[key], changes[key]))) return;

    this.state = { ...this.state, ...changes };
    this.version++;
    this.schedule();
  }

  schedule() {
    if (this.batch === "sync") {
      this.notify();
      return;
    }
    if (this.pending) return;
    this.pending = true;
    const run = () => {
      this.pending = false;
      this.notify();
    };
    if (this.batch === "raf") requestAnimationFrame(run);
    else queueMicrotask(run);
  }

  notify() {
    const state = this.state;
    const version = this.version;
    for (const listener of [...this.listeners]) {
      try {
        listener(state, version);
      } catch (error) {
        console.error("State listener failed:", error);
      }
    }
  }

  subscribe(listener) {
    if (typeof listener !== "function") {
      throw new TypeError("State.subscribe(): listener must be a function");
    }
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(item => item !== listener);
    };
  }
}

//EVENT_MAP (Translate) -> addEvent (Clean Old + Attach New) -> handleKey (Shortcuts) -> removeEvents (RAM Cleanup)
// Map framework props to native browser DOM events
const EVENT_MAP = {
  onClick: 'click',
  onInput: 'input',
  onChange: 'change',
  onKeyDown: 'keydown',
  onKeyUp: 'keyup',
  onSubmit: 'submit',
  onDblClick: 'dblclick',
  onBlur: 'blur',
  onFocus: 'focus'
};

// Remove one custom event listener from an element
export function removeEvent(element, propName) {
  const nativeEvent = EVENT_MAP[propName];

  if (!nativeEvent || !element._listeners) return;

  const handler = element._listeners[propName];

  if (!handler) return;

  element.removeEventListener(nativeEvent, handler);
  delete element._listeners[propName];

  if (Object.keys(element._listeners).length === 0) {
    delete element._listeners;
  }
}

// Attach custom event listener and prevent duplicate listeners
export function addEvent(element, propName, handler) {
  // Get native DOM event name (e.g., 'onClick' -> 'click')
  const nativeEvent = EVENT_MAP[propName];

  // Stop if the event name is unsupported
  if (!nativeEvent) return;

  // A missing/non-function handler means the prop was removed.
  if (typeof handler !== 'function') {
    removeEvent(element, propName);
    return;
  }

  // Create listener storage object if it does not exist
  if (!element._listeners) {
    element._listeners = {};
  }

  // Remove old event listener if it already exists
  if (element._listeners[propName]) {
    element.removeEventListener(nativeEvent, element._listeners[propName]);
  }

  // Save new handler in element storage
  element._listeners[propName] = handler;

  // Add new event listener to the native DOM element
  element.addEventListener(nativeEvent, handler);
}

// Remove all event listeners when element is deleted from DOM
export function removeEvents(element) {
  if (!element) return;

  // Loop through all saved event handlers
  for (const propName of Object.keys(element._listeners ?? {})) {
    const nativeEvent = EVENT_MAP[propName];
    if (nativeEvent) {
      // Remove each event listener from browser RAM
      element.removeEventListener(nativeEvent, element._listeners[propName]);
    }
  }

  // Delete storage property to completely clear memory
  delete element._listeners;
  for (const child of element.children ?? []) removeEvents(child);
}

export function listen(target, type, handler, options) {
  target.addEventListener(type, handler, options);
  return () => target.removeEventListener(type, handler, options);
}

// Helper function for handling keyboard shortcuts (Enter & Escape)
export function handleKey(e, onEnter, onEscape) {
  // Execute onEnter callback if Enter key is pressed
  if (e.key === 'Enter' && onEnter) {
    onEnter(e);
  // Execute onEscape callback if Escape key is pressed
  } else if (e.key === 'Escape' && onEscape) {
    onEscape(e);
  }
}

// public/src/net/socket.js
// Client WebSocket service. API contract: docs/ARCHITECTURE.md §7.2.
// No auto-reconnect on purpose: a refresh is a new session.

export function createSocket({ url } = {}) {
  const handlers = new Map();
  const statusHandlers = new Set();

  let ws = null;
  let status = "closed";

  const CONNECTING = 0;
  const OPEN = 1;

  function resolveUrl() {
    if (url) return url;
    const protocol = location.protocol === "https:" ? "wss" : "ws";
    return `${protocol}://${location.host}`;
  }

  function setStatus(next) {
    if (status === next) return;
    status = next;
    for (const callback of [...statusHandlers]) {
      try {
        callback(status);
      } catch (error) {
        console.error("[socket] status handler failed:", error);
      }
    }
  }

  function dispatch(type, data) {
    const callbacks = handlers.get(type);
    if (!callbacks) return;
    for (const callback of [...callbacks]) {
      try {
        callback(data);
      } catch (error) {
        console.error(`[socket] "${type}" handler failed:`, error);
      }
    }
  }

  function connect() {
    if (ws && (ws.readyState === CONNECTING || ws.readyState === OPEN)) return;

    setStatus("connecting");

    const socket = new WebSocket("ws://localhost:8080/");
    ws = socket;

    socket.onopen = () => {
      if (ws !== socket) return;
      setStatus("open");
    };

    socket.onmessage = (event) => {
      if (ws !== socket) return;

      let message;
      try {
        message = JSON.parse(event.data);
      } catch {
        return; // ignore malformed frames
      }

      if (
        !message ||
        typeof message !== "object" ||
        typeof message.type !== "string"
      )
        return;

      dispatch(message.type, message.data);
    };

    socket.onerror = () => {
      // The close event that follows drives the status change.
    };

    socket.onclose = () => {
      if (ws !== socket) return;
      ws = null;
      setStatus("closed");
    };
  }

  function send(type, data) {
    if (!ws || ws.readyState !== OPEN) {
      console.warn(`[socket] dropped "${type}": the connection is not open`);
      return;
    }
    try {
      ws.send(JSON.stringify({ type, data }));
    } catch (error) {
      console.warn(`[socket] could not send "${type}":`, error);
    }
  }

  function on(type, callback) {
    if (!handlers.has(type)) handlers.set(type, new Set());
    handlers.get(type).add(callback);

    return () => {
      const callbacks = handlers.get(type);
      if (!callbacks) return;
      callbacks.delete(callback);
      if (callbacks.size === 0) handlers.delete(type);
    };
  }

  function onStatus(callback) {
    statusHandlers.add(callback);

    try {
      callback(status); // called immediately with the current status
    } catch (error) {
      console.error("[socket] status handler failed:", error);
    }

    return () => {
      statusHandlers.delete(callback);
    };
  }

  function close() {
    const socket = ws;
    ws = null;
    if (socket) {
      try {
        socket.close();
      } catch {
        // Already closed.
      }
    }
    setStatus("closed");
  }

  return {
    connect,
    send,
    on,
    onStatus,
    close,
    get status() {
      return status;
    },
  };
}

/**
 * Initializes the FPS & Performance Overlay using requestAnimationFrame
 * @param {string} containerId - DOM element ID where FPS is rendered
 */
export function initFPSMeter(containerId = 'fps-meter') {
  const container = document.getElementById(containerId);
  if (!container) return;

  let frameCount = 0;
  let lastTime = performance.now();
  let fps = 60;
  let frameTime = 0;

  function tick(now) {
    frameCount++;
    const delta = now - lastTime;

    // Update FPS readout every 1 second (1000ms)
    if (delta >= 1000) {
      fps = Math.round((frameCount * 1000) / delta);
      frameTime = (delta / frameCount).toFixed(1);

      // Render readout text
      container.innerHTML = `FPS: ${fps} | ${frameTime}ms`;

      // Dynamic color coding based on performance threshold
      if (fps >= 55) {
        container.style.color = '#00ff66'; // Green = Good (60 FPS target)
      } else if (fps >= 40) {
        container.style.color = '#ffa502'; // Orange = Minor drops
      } else {
        container.style.color = '#ff4757'; // Red = Critical frame drops
      }

      frameCount = 0;
      lastTime = now;
    }

    // Schedule next frame measurement
    requestAnimationFrame(tick);
  }

  // Start the measurement loop
  requestAnimationFrame(tick);
}

  const off = listen(window, "framework-check", () => calls++);
  window.dispatchEvent(new Event("framework-check"));
  off(); off();
  window.dispatchEvent(new Event("framework-check"));
  assert(calls === 1, "cleanup removes listener");
});

await check("events replace and removed subtrees clean descendants", () => {
  const root = mount();
  let oldCalls = 0, newCalls = 0;
  const oldHandler = () => oldCalls++;
  const newHandler = () => newCalls++;
  render(h("section", {}, h("button", { onClick: oldHandler }, "click")), root);
  const button = root.firstChild.firstChild;
  button.click();
  render(h("section", {}, h("button", { onClick: newHandler }, "click")), root);
  button.click();
  assert(oldCalls === 1 && newCalls === 1, "handler replacement");
  render(h("main", {}, "replacement"), root);
  button.click();
  assert(newCalls === 1 && !button._listeners, "clean child even if removed parent has no listeners");
});

await check("chat log updates preserve an uncontrolled input", () => {
  const root = mount();
  const tree = messages => h("div", {},
    h("ul", {}, messages.map(text => h("li", {}, text))), h("input", {}));
  render(tree(["one"]), root);
  const input = root.querySelector("input"); input.value = "draft";
  render(tree(["one", "two"]), root);
  assert(root.querySelector("input") === input && input.value === "draft", "input survives log updates");
});

await check("form properties and fragment roots still work", () => {
  const root = mount();
  render([h("input", { value: "Ada", disabled: true }), h("span", {}, 3)], root);
  const input = root.firstChild;
  render([h("input", { value: "Bob", disabled: false }), h("span", {}, 2)], root);
  assert(root.firstChild === input && input.value === "Bob" && !input.disabled, "value and boolean attributes");
  assert(root.lastChild.textContent === "2", "fragment text update");
});
