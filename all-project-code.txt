// Framework Client-Side Router (Hash-based)

export function createRouter(routes = {}) {
  // Get current route path from URL hash (e.g. "#/active" -> "/active")
  function getPath() {
    const hash = window.location.hash.slice(1); // remove '#'
    return hash || '/'; // Default route if hash is empty
  }

  // Find matching route and run its callback function
  function handleRoute() {
    const currentPath = getPath();
    const routeCallback = routes[currentPath] || routes['/']; // Fallback to '/' if route not found
    
    if (typeof routeCallback === 'function') {
      routeCallback(currentPath);
    }
  }

  return {
    // Navigate programmatically to a new path (e.g., router.navigate('/active'))
    navigate(path) {
      window.location.hash = path;
    },

    // Start listening for URL hash changes and execute initial route
    start() {
      // Listen to browser hash changes (Back/Forward buttons or clicking links)
      window.addEventListener('hashchange', handleRoute);

      // Execute route callback immediately on page load
      handleRoute();
    }
  };
}

import normalizeChildren from "./normalize.js";

export  function h(type, props, ...children) {
  if (typeof type !== "string" || type.trim() === "") {
    throw new TypeError("h(): type must be a non-empty string");
  }

  if (props == null) {
    props = {};
  }

  if (typeof props !== "object" || Array.isArray(props)) {
    throw new TypeError("h(): props must be a plain object or null");
  }

  const prototype = Object.getPrototypeOf(props);

  if (prototype !== Object.prototype && prototype !== null) {
    throw new TypeError("h(): props must be a plain object or null");
  }

  return {
    type,
    props,
    children: normalizeChildren(children),
  };
}

function updateProps(element, oldProps = {}, newProps = {}) {
  const keys = new Set([...Object.keys(oldProps), ...Object.keys(newProps)]);

  for (const key of keys) {
    const oldValue = oldProps[key];
    const newValue = newProps[key];

    if (oldValue === newValue) continue;
    if (key === "key") continue;

    if (key.startsWith("on")) {
      addEvent(element, key, newValue);
      continue;
    }

    if (key === "value" && "value" in element) {
      const value = newValue == null ? "" : String(newValue);

      if (element.value !== value) {
        element.value = value;
      }

      continue;
    }

    if (key === "checked" && "checked" in element) {
      element.checked = Boolean(newValue);
      continue;
    }

    if (key === "selected" && "selected" in element) {
      element.selected = Boolean(newValue);
      continue;
    }
    if (newValue == null || newValue === false) {
      element.removeAttribute(key);
      continue;
    }

    if (newValue === true) {
      element.setAttribute(key, "");
      continue;
    }

    element.setAttribute(key, String(newValue));
  }
}

// Reconcile a list of children positionally inside `parent`.
function patchChildren(parent, oldChildren = [], newChildren = []) {
  const sharedLength = Math.min(oldChildren.length, newChildren.length);

  for (let i = 0; i < sharedLength; i++) {
    updateDOM(parent, newChildren[i], oldChildren[i], i);
  }

  for (let i = oldChildren.length - 1; i >= newChildren.length; i--) {
    const childToRemove = parent.childNodes[i];
    if (childToRemove) {
      removeEvents(childToRemove);
      parent.removeChild(childToRemove);
    }
  }

  for (let i = oldChildren.length; i < newChildren.length; i++) {
    parent.appendChild(createDom(newChildren[i]));
  }
}

function updateDOM(parent, newVNode, oldVNode, index = 0) {
  const current = parent.childNodes[index];

  if (newVNode == null) {
    if (current) {
      removeEvents(current); 
      parent.removeChild(current);
    }
    return;
  }

  if (oldVNode == null) {
    parent.insertBefore(createDom(newVNode), current ?? null);
    return;
  }

  const newIsPrimitive =
    typeof newVNode === "string" || typeof newVNode === "number";

  const oldIsPrimitive =
    typeof oldVNode === "string" || typeof oldVNode === "number";

  if (newIsPrimitive || oldIsPrimitive) {
    if (
      newIsPrimitive &&
      oldIsPrimitive &&
      typeof newVNode === typeof oldVNode
    ) {
      if (String(newVNode) !== String(oldVNode)) {
        current.nodeValue = String(newVNode);
      }
    } else {
      removeEvents(current); 
      parent.replaceChild(createDom(newVNode), current);
    }

    return;
  }

  if (newVNode.type !== oldVNode.type) {
    removeEvents(current); 
    parent.replaceChild(createDom(newVNode), current);
    return;
  }

  updateProps(current, oldVNode.props, newVNode.props);

  patchChildren(current, oldVNode.children ?? [], newVNode.children ?? []);
}

export default function patch(oldVNode, newVNode, container) {
  // A fragment root is a flat list of siblings inside the container.
  if (Array.isArray(oldVNode) || Array.isArray(newVNode)) {
    patchChildren(
      container,
      Array.isArray(oldVNode) ? oldVNode : [oldVNode],
      Array.isArray(newVNode) ? newVNode : [newVNode]
    );
    return;
  }

  updateDOM(container, newVNode, oldVNode, 0);
}

const previousVNodes = new WeakMap();

export  function render(vnode, container) {
  if (!(container instanceof Element)) {
    throw new TypeError("render(): container must be a valid DOM element");
  }

  if (!vnode || typeof vnode !== "object") {
    throw new TypeError("render(): vnode must be a valid VNode");
  }

  const oldVNode = previousVNodes.get(container);

  if (!oldVNode) {
    const dom = createDom(vnode);
    container.replaceChildren(dom);
  } else {
    patch(oldVNode, vnode, container);
  }

  previousVNodes.set(container, vnode);
}

export default function addprop(vnode, element) {
  if (!vnode.props) return element;

  for (const [key, value] of Object.entries(vnode.props)) {
    if (key === "key") continue;

    if (key.startsWith("on")) {

      continue;

    if (key === "value" && "value" in element) {
      element.value = value == null ? "" : String(value);
      continue;
    }

    if (key === "checked" && "checked" in element) {
      element.checked = Boolean(value);
      continue;
    }

    if (key === "selected" && "selected" in element) {
      element.selected = Boolean(value);
      continue;
    }

    // Handle standard HTML attributes
    if (value == null || value === false) {
      element.removeAttribute(key);
      continue;
    }

    if (value === true) {
      element.setAttribute(key, "");
    } else {
      element.setAttribute(key, String(value));
    }
  }

  return element;
}

export default function normalizeChildren(children) {

    .filter((child) => {
      return (
        child !== null && child !== undefined && typeof child !== "boolean"
      );
    })
    .map((child) => {
      if (typeof child === "number") {
        return String(child);
      }

      return child;
    });
}

import addprop from "./setProp.js";

export default function createDom(vnode) {
  // A fragment (array of vnodes) renders as sibling nodes with no wrapper element.
  if (Array.isArray(vnode)) {
    const fragment = document.createDocumentFragment();

    vnode.forEach((child) => {
      fragment.appendChild(createDom(child));
    });

    return fragment;
  }

  if (typeof vnode === "string" || typeof vnode === "number") {
    return document.createTextNode(vnode);
  }

  let element = document.createElement(vnode.type);

  if (vnode.props) {
     addprop(vnode, element);
  }

  if (vnode.children) {
    vnode.children.forEach((child) => {
      element.appendChild(createDom(child));
    });
  }

  return element;
}

export class State {
    constructor(initialState = {}) {
        this.state = initialState;
        this.listeners = [];

    }

    getState() {
        return this.state;
    }

    setState(update) {
        const changes = typeof update === "function" ? update(this.state) : update;
        this.state = { ...this.state, ...changes };
        this.listeners.forEach(listener => listener(this.state));
    }

    subscribe(listener) {
        this.listeners.push(listener);
        return () => {
            this.listeners = this.listeners.filter(ele => ele !== listener);
        }
    }
}

//EVENT_MAP (Translate) -> addEvent (Clean Old + Attach New) -> handleKey (Shortcuts) -> removeEvents (RAM Cleanup)
// Map framework props to native browser DOM events
const EVENT_MAP = {
  onClick: 'click',
  onInput: 'input',
  onChange: 'change',
  onKeyDown: 'keydown',
  onKeyUp: 'keyup',
  onSubmit: 'submit',
  onDblClick: 'dblclick',
  onBlur: 'blur',
  onFocus: 'focus'
};

// Remove one custom event listener from an element
export function removeEvent(element, propName) {
  const nativeEvent = EVENT_MAP[propName];

  if (!nativeEvent || !element._listeners) return;

  const handler = element._listeners[propName];

  if (!handler) return;

  element.removeEventListener(nativeEvent, handler);
  delete element._listeners[propName];

  if (Object.keys(element._listeners).length === 0) {
    delete element._listeners;
  }
}

// Attach custom event listener and prevent duplicate listeners
export function addEvent(element, propName, handler) {
  // Get native DOM event name (e.g., 'onClick' -> 'click')
  const nativeEvent = EVENT_MAP[propName];

  // Stop if the event name is unsupported
  if (!nativeEvent) return;

  // A missing/non-function handler means the prop was removed.
  if (typeof handler !== 'function') {
    removeEvent(element, propName);
    return;
  }

  // Create listener storage object if it does not exist
  if (!element._listeners) {
    element._listeners = {};
  }

  // Remove old event listener if it already exists
  if (element._listeners[propName]) {
    element.removeEventListener(nativeEvent, element._listeners[propName]);
  }

  // Save new handler in element storage
  element._listeners[propName] = handler;

  // Add new event listener to the native DOM element
  element.addEventListener(nativeEvent, handler);
}

// Remove all event listeners when element is deleted from DOM
export function removeEvents(element) {
  // Stop if element has no saved listeners
  if (!element || !element._listeners) return;

  // Loop through all saved event handlers
  for (const propName in element._listeners) {
    const nativeEvent = EVENT_MAP[propName];
    if (nativeEvent) {
      // Remove each event listener from browser RAM
      element.removeEventListener(nativeEvent, element._listeners[propName]);
    }
  }

  // Delete storage property to completely clear memory
  delete element._listeners;
}

// Helper function for handling keyboard shortcuts (Enter & Escape)
export function handleKey(e, onEnter, onEscape) {
  // Execute onEnter callback if Enter key is pressed
  if (e.key === 'Enter' && onEnter) {
    onEnter(e);
  // Execute onEscape callback if Escape key is pressed
  } else if (e.key === 'Escape' && onEscape) {
    onEscape(e);
  }
}

    console.log("data reconect is", data);
    router.navigate("/lobby");

  socketConnection.off("player:error");
  socketConnection.on("player:error", (message) => {
    error.textContent = message;

    "div",
    { className: "name-page" },
    h(
      "div",
      { className: "name-card" },

# Bomberman DOM

Multiplayer Bomberman-style game built with a custom DOM framework,
Node.js, Socket.IO, HTML, CSS, and JavaScript.

## Requirements
- 2–4 players per match; three lives each.
- Fixed grid with destructible blocks and indestructible walls.
- Bombs, explosions, and power-ups.
- Real-time multiplayer synchronization and chat.
- Last player alive wins.
- Efficient DOM rendering targeting 60 FPS.
- No Canvas, WebGL, React, Vue, or Angular.

## Structure
- `backend/`: Node.js and Socket.IO server.
- `frontend/`: custom DOM framework, game engine, UI, and rendering.
- `docs/`: shared technical documentation.

## Backend setup
```bash
cd backend
npm init -y
npm install socket.io
npm install --save-dev nodemon
```

The frontend should use a browser-compatible development server and install
`socket.io-client`.

1. Read `docs/SOCKET_PROTOCOL.md`.
2. Implement lobby and connection lifecycle.
3. Implement map and game engine.
4. Implement bombs, combat, and power-ups.
5. Integrate UI and test multiplayer gameplay.

function idPlayerIdTaken(playerId) {
  for (const player of players.values()) {
    if (player.playerId.toLowerCase() === playerId.toLowerCase()) {
      return true;
    }
  }

  return false;
}

export function registerPlayer(socket, nickname, playerId) {
  const name = validateNickname(nickname);

  if (!name) {
    return {
      success: false,
      message: "Invalid nickname",
    };
  }

  if (isNicknameTaken(name) || idPlayerIdTaken(playerId)) {
    return {
      success: false,
      message: "Nickname already taken",
    };
  }

  const playerToken = crypto.randomBytes(32).toString("hex");

  const player = {
    playerId,
    name,
    socketId: socket.id,
    connected: true,
    lives: 3,
  };

  players.set(playerId, player);
  tokens.set(playerToken, playerId);

  return {
    success: true,
    response: {
      player,
      playerToken,
    },
  };
}

// Find player by socket ID and mark disconnected
export function removePlayer(socketId) {
  for (const player of players.values()) {
    if (player.socketId === socketId) {
      player.connected = false;
      player.socketId = null;

      return true;
    }
  }
  return false;
}

export function getPlayers() {
  return Array.from(players.values()).map((player) => ({
    playerId: player.playerId,
    name: player.name,
    lives: player.lives,
    connected: player.connected,
  }));
}

export function StartCounterDown() {
  const connectedPlayers = Array.from(players.values()).filter(
    (player) => player.connected,
  );

  if (connectedPlayers.length >= 2) {
    return {
      success: true,
      counter: 10,
    };
  }

  return {
    success: false,
  };
}

# Socket.IO Protocol

Shared event contract for all developers. Keep event names and payload shapes
consistent across frontend and backend.

## General rules
- The server is authoritative for lobby and gameplay state.
- Clients send requests; the server validates them.
- Server events broadcast accepted state changes.
- Use consistent error formats and server-assigned player IDs.
- Never trust client-reported lives, positions, damage, or victory.

## Client → Server events

| Event | Purpose | Example payload |
|---|---|---|
| `player:join` | Request to join lobby | `{ "nickname": "Player1" }` |
| `chat:send` | Send chat message | `{ "message": "Hello" }` |
| `player:move` | Request movement | `{ "direction": "up", "sequence": 1 }` |
| `bomb:place` | Request bomb placement | `{ "x": 2, "y": 3 }` |

## Server → Client events

| Event | Purpose |
|---|---|
| `player:joined` | Join accepted |
| `player:error` | Join/action rejected |
| `lobby:update` | Players, count, capacity, state |
| `lobby:countdown` | Countdown value and state |
| `game:start` | Match started and initial state |
| `game:state` | Authoritative game-state update |
| `game:playerUpdated` | Player stats/status changed |
| `game:blockDestroyed` | Block destroyed and optional power-up |
| `game:explosion` | Explosion result |
| `game:playerEliminated` | Player eliminated |
| `game:over` | Winner or draw |
| `chat:message` | Chat message broadcast |
| `player:left` | Player disconnected or left |

## Example error payload

```json
{
  "code": "NICKNAME_TAKEN",
  "message": "That nickname is already in use."
}
```

## Example lobby update

```json
{
  "players": [{ "id": "server-player-id", "nickname": "Player1" }],
  "count": 1,
  "capacity": 4,
  "state": "WAITING"
}
```

## Example game state

```json
{
  "matchId": "server-match-id",
  "tick": 1,
  "players": [],
  "map": { "width": 15, "height": 13, "tiles": [] },
  "bombs": [],
  "powerUps": []
}
```

These are starter contracts. Agree on exact fields and validation before
implementation.

## Ownership
1. Developer 1: lobby lifecycle, chat transport, Socket.IO registration.
2. Developer 2: map representation and client rendering.
3. Developer 3: authoritative gameplay rules and match outcomes.
4. Developer 4: screens and frontend integration.

Communicate shared payload changes to all developers.

- Prevent duplicate active nicknames using case-insensitive comparison.
- Assign each connection a unique server-side player ID.
