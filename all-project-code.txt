import { test, mock } from "node:test";
import assert from "node:assert/strict";
import { State } from "../public/src/framework/state/index.js";

test("default notification stays synchronous and preserves unrelated state", () => {
  const store = new State({ count: 0, label: "keep" });
  const seen = [];
  store.subscribe((state, version) => seen.push([state, version]));
  store.setState({ count: 1 });
  assert.deepEqual(seen, [[{ count: 1, label: "keep" }, 1]]);
});

test("microtask batching merges immediately and notifies once with latest version", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const seen = [];
  store.subscribe((state, version) => seen.push([state.count, version]));
  store.setState({ count: 1 });
  store.setState(state => ({ count: state.count + 1 }));
  store.setState(state => ({ count: state.count + 1 }));
  assert.equal(store.getState().count, 3);
  assert.deepEqual(seen, []);
  await Promise.resolve();
  assert.deepEqual(seen, [[3, 3]]);
  store.setState({ count: 4 });
  await Promise.resolve();
  assert.deepEqual(seen, [[3, 3], [4, 4]]);
});

test("invalid/no-op patches preserve state, version and notification count", () => {
  const store = new State({ count: NaN, data: [] });
  const before = store.getState();
  const listener = mock.fn();
  store.subscribe(listener);
  for (const patch of [null, undefined, false, 1, "text", [], {}, { count: NaN }, { data: before.data }]) store.setState(patch);
  assert.equal(store.getState(), before);
  assert.equal(store.version, 0);
  assert.equal(listener.mock.callCount(), 0);
});

test("listener errors do not prevent other subscribers from running", t => {
  const log = t.mock.method(console, "error", () => {});
  const store = new State({ count: 0 });
  const listener = mock.fn();
  store.subscribe(() => { throw new Error("test listener"); });
  store.subscribe(listener);
  store.setState({ count: 1 });
  assert.equal(log.mock.callCount(), 1);
  assert.equal(listener.mock.callCount(), 1);
});

test("unsubscribe before scheduled notification removes the subscriber", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const listener = mock.fn();
  const off = store.subscribe(listener);
  store.setState({ count: 1 });
  off(); off();
  await Promise.resolve();
  assert.equal(listener.mock.callCount(), 0);
});

test("self-unsubscribe does not skip the next listener", () => {
  const store = new State({ count: 0 });
  const seen = [];
  const off = store.subscribe(() => { seen.push("first"); off(); });
  store.subscribe(() => seen.push("second"));
  store.setState({ count: 1 });
  store.setState({ count: 2 });
  assert.deepEqual(seen, ["first", "second", "second"]);
});

test("updates from microtask listeners schedule another notification", async () => {
  const store = new State({ count: 0 }, { batch: "microtask" });
  const seen = [];
  store.subscribe(state => {
    seen.push(state.count);
    if (state.count === 1) store.setState({ count: 2 });
  });
  store.setState({ count: 1 });
  await Promise.resolve();
  await Promise.resolve();
  assert.deepEqual(seen, [1, 2]);
});

test("raf batches and schedules again after notification", t => {
  const callbacks = [];
  const previous = Object.getOwnPropertyDescriptor(globalThis, "requestAnimationFrame");
  Object.defineProperty(globalThis, "requestAnimationFrame", {
    configurable: true, writable: true, value: callback => callbacks.push(callback),
  });
  t.after(() => {
    if (previous) Object.defineProperty(globalThis, "requestAnimationFrame", previous);
    else delete globalThis.requestAnimationFrame;
  });
  const store = new State({ count: 0 }, { batch: "raf" });
  const listener = mock.fn();
  store.subscribe(listener);
  store.setState({ count: 1 }); store.setState({ count: 2 });
  assert.equal(callbacks.length, 1);
  assert.equal(listener.mock.callCount(), 0);
  callbacks.shift()();
  assert.equal(listener.mock.callCount(), 1);
  assert.equal(listener.mock.calls[0].arguments[0].count, 2);
  store.setState({ count: 3 });
  assert.equal(callbacks.length, 1);
});

test("invalid batch modes and subscriber types fail clearly", () => {
  assert.throws(() => new State({}, { batch: "later" }), TypeError);
  assert.throws(() => new State().subscribe(null), TypeError);
});

# Bomberman DOM — team starter

A 2–4 player DOM Bomberman using our mini-framework and WebSockets.
This commit prepares the team to begin implementation; it is not a playable game.

## Start here

Use Node.js 20+ and npm. From the repository root:

```sh
npm ci
npm test
npm start
```

Open http://localhost:8080/ for the starter page and
http://localhost:8080/dev/framework-check.html for browser framework checks.
Use `PORT=8081 npm start` if 8080 is occupied.

With Chrome installed, `npm run test:browser` runs the same browser checks
automatically. For Chromium or a custom location, set `CHROME_BIN` to its
executable. This optional local headless check uses a temporary browser profile.

Included: a static server, page/CSS starter, shared constants, the patched runtime
framework, automated state/server checks, and browser verification checks.
WebSocket transport, nickname/lobby/chat, the game engine and game renderer are
still assigned work. `?fake=1` and `?debug` become available when their owners
implement them; they are not features of this starter.

1. [Objective and audit](bomberman-dom.md).
2. [Architecture, interfaces, and shared milestones](docs/ARCHITECTURE.md).
3. [WebSocket protocol](docs/PROTOCOL.md).

// Optional real-browser checks; Chrome/Chromium must be installed separately.
import { execFile } from "node:child_process";
import { once } from "node:events";
import { mkdtemp, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { promisify } from "node:util";
import { createAppServer } from "../server/index.js";

const server = createAppServer();
const profile = await mkdtemp(join(tmpdir(), "bomberman-chrome-"));
try {
  server.listen(0, "127.0.0.1");
  await once(server, "listening");
  const url = `http://127.0.0.1:${server.address().port}/dev/framework-check.html`;
  const { stdout } = await promisify(execFile)(process.env.CHROME_BIN ?? "google-chrome", [
    "--headless=new", "--no-sandbox", "--no-first-run", "--no-default-browser-check",
    "--disable-background-networking", `--user-data-dir=${profile}`,
    "--virtual-time-budget=2000", "--dump-dom", url,
  ], { timeout: 20000, maxBuffer: 2 * 1024 * 1024 });
  const report = stdout.match(/<pre id="out">([\s\S]*?)<\/pre>/)?.[1];
  if (!/<html[^>]*data-status="passed"/.test(stdout)) {
    throw new Error(report ?? "Browser checks did not complete; inspect /dev/framework-check.html");
  }
  console.log(report);
} catch (error) {
  console.error(error.code === "ENOENT"
    ? "Chrome was not found. Set CHROME_BIN to your Chrome/Chromium executable, or open the check page manually."
    : error.message);
  process.exitCode = 1;
} finally {
  if (server.listening) await new Promise(resolve => server.close(resolve));
  await rm(profile, { recursive: true, force: true });
}

# TASKS — mohamed: game engine (server-side rules, pure module)

Read first: `ARCHITECTURE.md` (§4, §5, §7.1), `PROTOCOL.md` (the `snapshot` and `tiles` messages), `../shared/constants.js` (from these docs; use `../../shared/constants.js` from `server/game/`).

## Mission

Write the **authoritative game simulation**: map generation, smooth movement with collisions, bombs, explosions, power-ups, lives, win/draw. It is a **pure Node module**: no sockets, no timers, no `Date.now()` (time only comes from `step(dt)`), no randomness except the seeded RNG. yacobe's room calls it 30 times per second; hichame renders what it returns.

## Boundaries

- You own `server/game/` and its tests. You never import `ws`, `http`, or anything from `server/net` / `server/lobby`.
- External imports are limited to `../../shared/constants.js` from `server/game/`; importing your own engine modules is allowed.
- The public API is shared (`ARCHITECTURE.md` §7.1). Coordinate corrections during M1, then stabilize it. Internals are yours.

## Files to create

```

server/game/map.js            generateMap(seed) -> { tiles: Uint8Array, hidden: Map<cellIndex, TILE.POWERUP_*> }
server/game/player.js         createPlayer(slot, nickname): stats + derived speed
server/game/movement.js       movePlayer(game, player, dtSec): collision + corner assist
server/game/bombs.js          placeBomb, updateBombs, explode (chain), updateFlames
server/game/powerups.js       pickupPowerup, revealHidden
server/game/damage.js         applyFlameDamage, eliminate
server/game/game.js           createGame: public API, tick order, snapshot/result
server/__tests__/map.test.js
server/__tests__/movement.test.js
server/__tests__/bombs.test.js
server/__tests__/game.test.js
```

## Public API (integration contract)

```js
createGame(players, seed) -> game       // players: [{ slot, nickname }], slots are 0..n-1
game.setInput(slot, dir)                // "up"|"down"|"left"|"right"|null   (ignore invalid / dead players)
game.placeBomb(slot)
game.removePlayer(slot)                 // lives = 0, alive = false (bombs already placed still explode)
game.step(dtMs) -> { tilesChanged }     // dtMs is normally TICK_MS; if > 100 split into sub-steps
game.getSnapshot() -> { tick, players: [ { slot, x, y, dir, moving, alive, lives, invulnerable, maxBombs, range, speedLevel } ] }   // x,y rounded to 3 decimals
game.getTiles() -> number[]             // COLS*ROWS TILE codes. Hidden power-ups are NEVER included (a BLOCK is a BLOCK).
game.getResult() -> null | { winner: slot|null, players: [ { slot, nickname, lives } ] }
```

## Features in detail

### 1. Map generation (`map.js`) — deterministic for a given seed
1. Border = `WALL`. Pillars: `WALL` at every interior `(x even, y even)`. Everything else `FLOOR`.
2. Safe zones for **all 4 corners** (even with fewer players): corner cell + `SAFE_ZONE_ARM` (3) cells along each edge, e.g. top-left: (1,1),(2,1),(3,1),(4,1),(1,2),(1,3),(1,4). Never a block there. Require `SAFE_ZONE_ARM > BASE_RANGE`; verify each corner has a reachable cell outside its first bomb blast, with enough time to reach it at base speed before the fuse ends. The former two-cell arms were entirely inside the default blast.
3. Every other `FLOOR` cell becomes `BLOCK` with probability `BLOCK_DENSITY` (seeded RNG).
4. **Hidden power-ups:** choose `max(round(blocks * POWERUP_CHANCE), 3 * POWERUP_MIN_PER_TYPE)` blocks. Build the deck so each of the 3 types appears at least `POWERUP_MIN_PER_TYPE` times, fill the rest randomly, shuffle (seeded), assign to the chosen blocks in a `Map<cellIndex, tileCode>`. This guarantees the audit question "at least 3 types of power-ups". Ensure at least six blocks exist before assigning the deck (deterministically add blocks to eligible cells if necessary); never select more hidden-item cells than actual blocks.

### 2. Players (`player.js`)
- Start at the centre of `CORNER_BY_SLOT[slot]` (e.g. `(1.5, 1.5)`), `dir: "down"`, `lives: LIVES`, `maxBombs: BASE_BOMBS`, `range: BASE_RANGE`, `speedLevel: 0`, `alive: true`, `invulnerableUntil: 0`, `activeBombs: 0`.
- `speed = BASE_SPEED + speedLevel * SPEED_STEP` tiles/s.

### 3. Movement (`movement.js`) — smooth, 4-directional, tile-based collision
- Input = desired direction (or null). Move `speed * dt` tiles along it. Split the movement into sub-steps of ≤ 0.25 tile so nobody tunnels through a wall.
- Hitbox = square centred on `(x, y)` with half-size `PLAYER_HITBOX_HALF` (0.35). A move is allowed only if every cell overlapped by the hitbox at the new position is **not solid**. Solid = `WALL`, `BLOCK`, `BOMB` (see exception below). Do the move per axis (only one axis moves at a time since there is no diagonal).
- **Corner assist** (mandatory, otherwise cornering feels awful): if the move is blocked, *and* the cell straight ahead in the player's centred lane is free, *and* the player's offset from the lane centre is ≤ `CORNER_ASSIST` (0.3), slide the player perpendicular toward the lane centre (by the same distance, never overshooting the centre) instead of stopping. Example: moving right at `y = row + 0.5 + 0.2` next to a pillar → nudge `y` toward `row + 0.5` until the move passes.
- **Own-bomb exception:** when a bomb is placed, every player whose hitbox overlaps that cell may pass through it (`bomb.passable` = set of slots). A slot is removed from the set as soon as its hitbox no longer overlaps the bomb cell; after that the bomb is solid for everybody.
- `dir` field = last non-null direction. `moving` = the position really changed this tick (blocked by a wall → false).
- Dead players don't move, don't collide, don't take damage.

### 4. Bombs (`bombs.js`)
- `placeBomb(slot)`: alive, `activeBombs < maxBombs`, the cell under the player's **centre** is `FLOOR` → tile becomes `BOMB`; store `{ ownerSlot, cell, range: player.range (captured now), explodeAt: now + BOMB_FUSE_MS, passable }`; `activeBombs++`.
- Every tick: bombs whose `explodeAt <= now` explode. Use a queue so **chain reactions** resolve in the same tick.
- **Explosion** from cell `c` with `range r`: `c` becomes `FLAME`. For each of the 4 directions, up to `r` cells:
  - `WALL` → stop (no flame);
  - `BLOCK` → becomes `FLAME`, remember to reveal its hidden power-up when the flame ends, **stop**;
  - `BOMB` (other bomb) → add to the explode queue, **stop**;
  - exposed `POWERUP_*` → destroyed, becomes `FLAME`, **continue**;
  - `FLOOR` / `FLAME` → becomes `FLAME`, continue.
  - Each flame cell stores `expiresAt = now + FLAME_MS` (take the max when overlapping).
- Explosion: `owner.activeBombs--`, `tilesChanged = true`. Process each bomb at most once, even when several rays queue it during one tick.
- Flame expiry: cell becomes the revealed power-up tile if one was hidden there, else `FLOOR`.

### 5. Power-ups (`powerups.js`)
- Each tick, for every alive player, look at the cell of the centre: if it is `POWERUP_BOMB/FLAME/SPEED` → apply and set the tile to `FLOOR`, `tilesChanged = true`:
  - bomb: `maxBombs + 1` (max `MAX_BOMBS`), flame: `range + 1` (max `MAX_RANGE`), speed: `speedLevel + 1` (max `MAX_SPEED_LEVEL`). Over the cap: tile is still consumed, no effect.

### 6. Damage, lives, elimination (`damage.js`)
- After bombs/flames are updated: for each alive, **non-invulnerable** player whose centre cell is `FLAME` → `lives--`, `invulnerableUntil = now + INVULN_MS`. At `lives === 0` → `alive = false`.
- `invulnerable` in the snapshot = `now < invulnerableUntil`.
- **This must hold:** standing on a bomb when it explodes costs exactly **1 life** (the flame lasts 15 ticks; invulnerability prevents more). A player can hurt himself. Another player next to the bomb also loses 1.

### 7. Tick order (`game.step`)
1. apply nothing for dead players; 2. move players; 3. pick up power-ups; 4. bomb fuses + chain explosions; 5. flame expiry (+ reveal); 6. damage; 7. eliminations; 8. `tick++`. Return `{ tilesChanged }`. Maintain a persistent tile-dirty flag: a `placeBomb()` call between steps must make the next step return true. OR together changes across internal sub-steps. At step end, capture the flag, clear the internal flag, then return the captured value; getters never clear it.

### 8. Result (`getResult`)
- `null` while ≥ 2 players are alive. When ≤ 1 alive: `winner` = that slot, or `null` if nobody is alive (draw). `players` = all players with their final `lives`. Works also right after `removePlayer`.

# PROTOCOL.md — WebSocket protocol (v0.2 INTEGRATION DRAFT)

**Owner: yacobe.** Review this draft with affected owners during M1 (two real browsers joining/chatting), then stabilize it. Contract changes update this file, consumers, and `shared/constants.js` when constants change, in the same PR. Constants live at `../shared/constants.js` relative to this document.

## 1. Transport

- One WebSocket per browser tab, same host/port as the HTTP server: `ws://${location.host}` (use `wss` if the page is https).
- Text frames, JSON only. Envelope: `{ "type": "<string>", "data": { ... } }`.
- TCP + one socket = messages arrive **in order**. The server relies on that: `start` always comes before the first `snapshot`, and a `tiles` message is sent before the `snapshot` of the same tick.
- Configure `WebSocketServer` with `maxPayload: MAX_MESSAGE_BYTES` (1024) and `perMessageDeflate: false`. Oversized messages close that connection with WebSocket code 1009; no JSON error response is required. Reject binary messages with code 1003. Also retain byte-length validation in the application parser. More than `MAX_MESSAGES_PER_SECOND` per socket → `RATE_LIMITED`, message dropped.
- The server never trusts the client: it only accepts the 4 message types below, validates every field, and ignores anything not allowed in the current phase.
- 10 `BAD_MESSAGE` errors from one socket → the server closes it.
- Every socket has an `error` handler. Server heartbeat: ping every 30 s, mark alive on pong, terminate when the previous ping was unanswered, and invoke normal disconnect cleanup exactly once. Browser clients reply to protocol pings automatically. Clear heartbeat timers on server shutdown.

## 2. Client → Server

| type | data | Allowed when | Notes |
|---|---|---|---|
| `join` | `{ "nickname": "Ada" }` | not joined, room phase `lobby` | trim; 2–12 chars; `NIKC_REGEX`; unique case-insensitively among joined players |
| `chat` | `{ "text": "hello" }` | joined (lobby or playing, dead players too) | trim; empty → ignored silently; > 200 chars → `BAD_MESSAGE`; max 5 per 5 s per player → `RATE_LIMITED`; control characters stripped |
| `input` | `{ "dir": "up" \| "down" \| "left" \| "right" \| null }` | joined, phase `playing` | **sent only when the effective direction changes**. `null` = stop. Last-pressed key wins (client keeps the held-keys stack). |
| `bomb` | `{}` (data may be omitted) | joined, phase `playing` | one-shot event; the engine decides if it is legal |

## 3. Server → Client

### `hello`
Sent right after the socket opens, and to everyone again after a game ends and the room resets.
```json
{ "type": "hello", "data": { "phase": "lobby", "count": 2, "max": 4 } }
```
`phase`: `"lobby"` (joining possible) or `"playing"` (join will be refused).

### `joined`
```json
{ "type": "joined", "data": { "nickname": "Ada" } }
```

### `error`
```json
{ "type": "error", "data": { "code": "NICK_TAKEN", "message": "Nickname already used" } }
```
Codes: `BAD_MESSAGE, UNKNOWN_TYPE, NICK_INVALID, NICK_TAKEN, ROOM_FULL, GAME_IN_PROGRESS, ALREADY_JOINED, NOT_JOINED, RATE_LIMITED`. `message` is human readable (the UI may show it directly).

### `lobby`
Sent to all **joined** players on every membership change **and once per second while a timer runs**.
```json
{ "type": "lobby", "data": {
    "players": ["Ada", "Bob"], "count": 2, "max": 4,
    "timer": { "kind": "wait", "secondsLeft": 14 } } }
```
`timer` is `null` when no timer runs. `kind`: `"wait"` (20 s, waiting for more players) or `"start"` (10 s "get ready" countdown). `secondsLeft = ceil(remainingMs / 1000)`.

### `chat`
```json
{ "type": "chat", "data": { "from": "Ada", "text": "hi", "ts": 1730000000000 } }
```
`from: null` = system message (`"Bob joined"`, `"Bob left"`, `"Bob was eliminated"`...). Sent to all joined players. No history for late joiners.

### `start`
Sent once, to every player, when the countdown ends.
```json
{ "type": "start", "data": {
    "you": 0, "cols": 15, "rows": 13,
    "players": [ { "slot": 0, "nickname": "Ada" }, { "slot": 1, "nickname": "Bob" } ],
    "tiles": [ 1,1,1, "... COLS*ROWS ints ..." ] } }
```
`you` = the receiver's slot. Slots are compacted to `0..n-1` in join order. Corner = `CORNER_BY_SLOT[slot]`.

### `tiles`
Full tile array, sent **only when something changed** (bomb placed, explosion, flame expiry, block destroyed, power-up revealed/picked up). Never a diff — a client that applies it can never be out of sync.
```json
{ "type": "tiles", "data": { "tiles": [ "... COLS*ROWS ints ..." ] } }
```

### `snapshot`
Every tick (30 Hz) while playing.
```json
{ "type": "snapshot", "data": { "tick": 1234, "players": [
  { "slot": 0, "x": 1.5, "y": 1.5, "dir": "down", "moving": false,
    "alive": true, "lives": 3, "invulnerable": false,
    "maxBombs": 1, "range": 2, "speedLevel": 0 } ] } }
```
- `x`,`y` = centre of the player in **tile units** (tile `(cx,cy)` covers `[cx, cx+1) × [cy, cy+1)`, so the centre of tile (1,1) is `(1.5, 1.5)`), rounded to 3 decimals.
- `dir` = facing direction (the last non-null direction); `moving` = position changed this tick.
- Dead players stay in the array with `alive: false, lives: 0`.
- `tick` starts at 0; send an initial snapshot immediately after `start`, then increment once per fixed simulation step. Use `tick * TICK_MS` as simulation time when interpolating. Server catch-up may deliver snapshots close together; receive-time spacing is not simulation duration.
- Baseline rendering delays player positions by 66 ms and applies tiles/status changes immediately. This can briefly show a delayed position beside a newly changed tile; gameplay/damage remain authoritative on the server. Test the visual timing and control latency during M3 before introducing prediction or a tick-buffered world renderer.

### `gameover`
```json
{ "type": "gameover", "data": { "winner": 2, "players": [
  { "slot": 0, "nickname": "Ada", "lives": 0 }, { "slot": 2, "nickname": "Cy", "lives": 1 } ] } }
```
`winner: null` = draw (everybody eliminated in the same tick). After sending it the server **resets**: phase `lobby`, nobody is joined any more, then sends `hello` to every socket. To play again a client sends `join` again.

## 4. Lobby rules (server implements, UI displays)

| Situation | Result |
|---|---|
| count reaches 2 | start `wait` timer (20 s) |
| `wait` running, count reaches 4 | switch **immediately** to `start` timer (10 s) |
| `wait` reaches 0 (2 or 3 players) | `start` timer (10 s) |
| `start` reaches 0 | game starts with the players present |
| join during `wait`/`start` (count < 4) | allowed, **timer is not reset** |
| count drops below 2 at any time before the start | timer cancelled (`timer: null`); when count reaches 2 again a fresh `wait` starts |
| `join` when count is 4 | `ROOM_FULL` |
| `join` while a game runs | `GAME_IN_PROGRESS` |

## 5. Disconnects

- In the lobby: the player is removed, `lobby` is re-broadcast.
- Client-side connection close during a match: clear held keys, cancel the game rAF, show a connection error on the nickname screen; refresh creates a new session. Do not keep animating stale gameplay.
- Mid-game: the player is **eliminated** (`removePlayer`). If one player is left alive, the game ends. Refreshing the page = disconnect = eliminated (accepted).

## 6. Sequence (happy path)

```
client                         server
  | -- connect ----------------> |
  | <------------------ hello ---|  {phase:"lobby", count:1}
  | -- join {Ada} -------------> |
  | <----------------- joined ---|
  | <------------------ lobby ---|  every change + each second of a timer
  | <------------------- chat ---|
  | <------------------ start ---|  tiles + slots
  | <-------------- snapshot x N-|  30 Hz
  | -- input {dir:"left"} -----> |
  | -- bomb -------------------> |
  | <------------------ tiles ---|  when the map changes
  | <---------------- gameover --|
  | <------------------ hello ---|  room reset
```

## Phase 3 — room and game loop

**`loop.js`**: `createLoop(stepFn, tickMs)`; uses `process.hrtime.bigint()` and an accumulator, runs `stepFn` for every full tick elapsed (cap catch-up at e.g. 5 ticks), scheduled with `setTimeout` ~ every 5 ms. A bare `setInterval(…, 33)` drifts and makes movement jerky: don't use it.

**`room.js`** — the only module that knows sessions, lobby *and* engine:
- `phase`: `"lobby"` or `"playing"`.
- `handle(session, type, data)`:
  - `join`: phase `playing` → `GAME_IN_PROGRESS`; already joined → `ALREADY_JOINED`; else `lobby.join`; on success send `joined`, broadcast `lobby`, broadcast chat system message `"<nick> joined"`.
  - `chat`: must be joined (`NOT_JOINED`), rate limit, sanitize (trim, strip control chars, ≤ 200) → broadcast `chat {from, text, ts}` to all joined sessions.
  - `input` / `bomb`: only joined + phase `playing`; map session → `slot`; call `game.setInput(slot, dir)` / `game.placeBomb(slot)`. Otherwise ignore.
- A 100 ms timer calls `lobby.update()`; broadcast `lobby` when `JSON.stringify(view)` changed (that covers the per-second timer ticks). On `{event:"start"}`:
  1. assign slots `0..n-1` in join order;
  2. `game = createGame(players.map(({nickname}, slot) => ({ slot, nickname })), Date.now())`;
  3. send each player `start { you, cols, rows, players, tiles: game.getTiles() }`, immediately followed by the initial `snapshot` (tick 0);
  4. phase = `playing`, start the loop.
- Each tick: `const { tilesChanged } = game.step(TICK_MS)`; if `tilesChanged` broadcast `tiles` **before** the snapshot; broadcast `snapshot`; `const result = game.getResult()`; if not null → broadcast `gameover`, stop the loop, **reset**: un-join every session, new empty lobby, phase `lobby`, send `hello` to every connected socket.
- Disconnect: lobby phase → `lobby.leave`, broadcast `lobby` + system chat; playing phase → `game.removePlayer(slot)` (+ system chat `"<nick> left"`). A disconnect that leaves < 2 players in the lobby cancels the timer (handled by the lobby).
- On connect (any phase) send `hello {phase, count, max}`.

# bomberman-dom — ARCHITECTURE (v1.1)

Read order: root `README.md` → `../bomberman-dom.md` (objective + audit) → this file → `PROTOCOL.md` → your `TASKS_<NAME>.md`.

Goal: a multiplayer Bomberman (2–4 players) in the browser, built with **our mini-framework only**, WebSockets, 60 fps, minimal paint/layers.

## 1. Hard constraints (from the subject / audit)

- Client: our mini-framework only. **No canvas, no WebGL, no other framework/library.**
- WebSockets for chat and gameplay. `requestAnimationFrame` for rendering. 60 fps, no frame drops (checked in DevTools Performance, with Paint flashing and Layer borders).
- Flow: nickname page → waiting page (player counter + chat) → 20 s wait (2+ players) → 10 s countdown (immediately if 4 players) → game.
- 3 lives, corners, fixed walls + random blocks, ≥ 3 power-up types (bomb+1, flame+1, speed), last one standing wins.
- Bonus features: **skipped**.

## 2. Principles

1. **The server is authoritative.** It owns positions, bombs, flames, blocks, power-ups, lives, win. Clients send inputs and render what they receive.
2. **Two data paths on the client.** Anything that changes more than ~10×/second (player positions) never goes through `State`. `State` is only for slow things (screen, lobby, chat, result, errors).
3. **Stable DOM.** The framework diffs positionally, so we never render growing/shrinking lists of game entities. The map is a fixed grid of 195 cells whose *class* changes; players are 4 fixed slots.
4. **Explicit ownership.** Follow the file/directory owners below; styles and server tests are shared directories with file-level ownership. Coordinate edits to `shared/constants.js` and `PROTOCOL.md` with Yacobe and all affected owners.
5. **Contracts first.** Code against section 7. Correct gaps together during M1, update the documents and consumers in the same PR, and stabilize after real integration. Avoid private workarounds.

## 4. Conventions

- **Grid:** 15 columns × 13 rows. Cell `(x, y)`, `x` right, `y` down, origin top-left. Tile index = `y * COLS + x`.
- **Walls:** border + pillars at `(x even, y even)`. **Blocks:** random on the remaining floor, never in a corner safe zone.
- **Safe zone** (always empty): corner cell + `SAFE_ZONE_ARM` (3) cells along each edge, for all 4 corners (even in 2-player games).
- **Escape invariant:** `SAFE_ZONE_ARM > BASE_RANGE`; from each spawn, an empty route must reach a cell outside the first bomb's blast before `BOMB_FUSE_MS`, at `BASE_SPEED`. Recheck this when tuning constants.
- **Positions** are floats in *tile units*, **centre-based**: the centre of cell (1,1) is (1.5, 1.5). Sprite top-left = `(x - 0.5, y - 0.5)` tiles.
- **Slots:** `0..3`, compacted in join order at game start. Corners: slot 0 TL, 1 BR, 2 TR, 3 BL. Colours: `--p0..--p3` (see 7.6).
- **Directions:** `"up" | "down" | "left" | "right" | null`. 4-directional movement only.
- **Time:** the server simulates at 30 Hz (`TICK_MS`). Clients render at display rate (60 fps) by interpolating between snapshots.

| Thing | Value |
|---|---|
| Lives | 3, with 2 s invulnerability after a hit (blinks) |
| Bomb fuse / flame duration | 2 s / 0.5 s |
| Base bombs / range / speed | 1 bomb / 2 tiles / 3.5 tiles per second |
| Power-up effects | bomb +1 (max 5), flame +1 (max 6), speed +0.6 (max 3 levels) |
| Blocks | 70 % of eligible floor cells; 30 % of blocks hide a power-up, **pre-assigned at map generation**, with ≥ 2 of each of the 3 types guaranteed |
| Damage | decided by the cell of the player's **centre**; a player can hurt himself; flames trigger other bombs (chain) |
| Flames | stop at walls; destroy the first block they hit (and stop); destroy exposed power-ups and continue through them; ignite bombs and stop that ray (the triggered bomb emits its own rays) |

## 6. Client performance contract

1. One `requestAnimationFrame` loop, running only during the game. No `setInterval` for rendering.
2. Per frame the framework should only diff ~4 player nodes. The tile grid vnode and the HUD/chat vnodes are **cached** and reused by reference (the framework patch makes `patch` skip identical vnodes). **A vnode is immutable once rendered.**
3. Only `transform` and `opacity` are animated. No CSS `transition` on position/size. No box-shadow/filter/blur on the board.
4. `will-change: transform` on the 4 player sprites only — **never on the 195 cells**. Inspect actual layer counts in DevTools; expect a small number, not one per cell. Browser layer assignment is measured, not guaranteed by this layout.
5. No layout reads (`offsetWidth`, `getBoundingClientRect`…) inside the frame loop. The board scales with CSS only (`--tile` from `min(vw, vh)`).
6. Event handlers in per-frame trees are **module-level constants** (never inline arrows), otherwise listeners are re-registered every frame.
7. No `innerHTML` anywhere. No `<svg>` elements (the framework has no SVG namespace support): sprites are CSS shapes or `<img>`.
8. Budget: under ~4 ms of JS per frame. `?debug` shows an FPS / frame-time overlay.

## 7. Interfaces (integration draft; stabilize after M1)

### 7.1 Game engine (`server/game/game.js`, owner mohamed) — pure, no sockets, no timers
```js
createGame(players, seed) -> game
//   players: [{ slot: 0..3, nickname }]   seed: integer (deterministic RNG)
game.setInput(slot, dir)        // dir: "up"|"down"|"left"|"right"|null
game.placeBomb(slot)
game.removePlayer(slot)         // disconnect => eliminated (lives 0, alive false)
game.step(dtMs) -> { tilesChanged: boolean }
game.getSnapshot() -> { tick, players: [ { slot, x, y, dir, moving, alive, lives, invulnerable, maxBombs, range, speedLevel } ] }
game.getTiles()    -> number[]   // length COLS*ROWS, TILE codes, hidden power-ups NOT included
game.getResult()   -> null | { winner: slot|null, players: [ { slot, nickname, lives } ] }
```
The room (yacobe) owns timing: it calls `step(TICK_MS)` at 30 Hz, then broadcasts. Engine mutations outside `step`, including `placeBomb`, set a persistent tile-dirty flag. At the end of the next `step`, capture that flag OR any tile changes during the tick, clear the internal flag, and return the captured value. Reading tiles/snapshots must not clear it.

### 7.2 Client socket (`public/src/net/socket.js`, owner yacobe)
```js
const socket = createSocket();            // or createFakeSocket() when the URL has ?fake=1
socket.connect();
socket.send(type, data);                  // JSON-encodes {type, data}; silently dropped (with console.warn) if not open
const off = socket.on(type, (data) => {}); // handler receives `data` only; returns an unsubscribe fn
const offStatus = socket.onStatus((status) => {});          // "connecting" | "open" | "closed"; called immediately with the current status
socket.close();
```
`createFakeSocket()` has the same API plus `emit(type, data)` and plays a scripted scenario (see TASKS_YACOBE).

### 7.3 Store (`public/src/state/store.js`, owner ayoub) — a framework `State`
```js
{
  connection: "connecting" | "open" | "closed",
  phase: "lobby" | "playing",              // from `hello`; set to playing at `start`
  screen: "nickname" | "lobby" | "game" | "over",
  me: null | { nickname, slot },           // slot is set at `start`
  error: null | { code, message },
  lobby: { players: [], count: 0, max: 4, timer: null | { kind, secondsLeft } },
  chat: [ { from, text, ts } ],            // from === null -> system. Immutable updates only. Capped at 100.
  players: [ { slot, nickname } ],         // from `start`
  result: null | { winner, players: [ { slot, nickname, lives } ] }
}
```
Create the app store with `new State(initialState, { batch: "microtask" })`. The framework default remains synchronous for compatibility. Never put per-frame data (positions) in the store.

### 7.4 Game view (`public/src/game/gameView.js`, owner hichame) — plain mutable module, NOT in the store
```js
gameView.reset(startData)           // from `start`: cols, rows, tiles, players, you
gameView.setTiles(tiles)            // from `tiles`
gameView.pushSnapshot(snapshot)     // from `snapshot`, stamps the receive time
gameView.getFrame(nowMs) -> { tiles, tilesVersion, players: [ ...interpolated players... ] }
```
`tilesVersion` increments on reset and when new tiles arrive (the renderer rebuilds the grid vnode only then). Retain a bounded `SNAPSHOT_HISTORY_MS` (500 ms) history of whole snapshots; two snapshots are insufficient for the 66 ms interpolation delay. Use server `tick * TICK_MS` for simulation timestamps and a local clock mapping. See Hichame's task for startup/stale-buffer behaviour.

### 7.5 Who handles which message, and who renders `#app`

| Message types | Handler module | What it does |
|---|---|---|
| `hello, joined, error, lobby, chat` | `ui/netHandlers.js` (ayoub) | updates the store |
| `start, tiles, snapshot, gameover` | `game/index.js` (hichame) | updates `gameView`, starts/stops the loop, sets `screen: "game"` / `"over"` in the store |

- `main.js` creates the socket, calls `initUI({ socket, store })` and `initGame({ socket, store })`, then `socket.connect()`.
- **Rendering of `#app`:** ayoub's `ui/render.js` subscribes to the store and renders the current screen **only when `state.screen !== "game"`**. While `screen === "game"`, hichame's rAF loop renders `#app` every frame (including the chat panel, built with ayoub's `chatPanel()` and cached by the `chat` array reference). Handover happens by `screen` changes only. Before leaving the game, cancel its pending rAF and clear held input. `start()` is idempotent; `stop()` cancels the stored rAF id. A closed socket during a match stops the loop, clears input, sets a connection error, and returns to the nickname screen. `onStatus` returns an unsubscribe function.
- `initUI` is exported from `ui/index.js`; `initGame` from `game/index.js`. Both return a cleanup function that unsubscribes handlers/listeners. Chat focus is observed by game input with `listen(document, "focusin", ...)`: clear held keys and send stop before typing. `hello` after `gameover` updates room phase but preserves the result screen.

## bomberman-dom

You certainly know [bomberman](https://en.wikipedia.org/wiki/Bomberman) right? Good good. You will make it. Relax, is not that hard, it is only a multiplayer version of it. Ah and forgot to mention that you will need to do it using the framework you created a while ago. Let me explain.

### Objectives

For this project you have to create a bomberman alike game, where multiple players can join in and battle until one of them is the last man standing.

### Instructions

In the beginning there are 4 players, and only one came out alive. Each player will have to start in the different corners of the map and only one will be victorious.

You will have to follow more or less the same principles of [make-your-game](../make-your-game/README.md) project. But we will refresh one of the concepts you will have to respect and deal with:

- [**Performance**](../good-practices/README.md), is one of the most important aspects while developing a game, so let's respect it.\
   Just like make-your-game you will have to respect the policy of:

  - Running the game at least at **60fps** at all time
  - No frame drops
  - Proper use of [**`requestAnimationFrame`**](https://developer.mozilla.org/en-US/docs/Web/API/window/requestAnimationFrame)
  - Measuring performance to know if your code is fast

You must not use canvas, neither [Web-GL](https://get.webgl.org/) nor another framework. For this project you will use the framework you did on the [mini-framework](../mini-framework/) project.

You will also have to make a chat that enables the different players to talk to each other. You will have to use [WebSockets](https://developer.mozilla.org/en-US/docs/Web/API/WebSockets_API). This chat **can be considered as a "Hello World" of the multiplayer feature** for the **bomberman-dom**.

#### Game Mechanics

1. Players

   1. Nº of players: 2 - 4
   2. Each player must have 3 lives. Then you are out!!

2. Map

   1. The map should be fixed so that every player sees the whole map.
   2. There will be two types of blocks, the ones that can be destroyed (blocks) and the ones that can not (walls).
      1. The walls will always be placed in the same place, while the blocks are meant to be generated randomly on the map. Tip: the optional project [different maps](../make-your-game-different-maps/README.md) can be useful for this part.
      2. In the starting positions the players need to be able to survive. For example: if the players place a bomb, they will need to have space to avoid the bomb explosion.
   3. The players should be placed in the corners as their starting positions.

3. Power ups (each time a player destroys a block, a random power up may or may not appear):

   1. Bombs: Increases the amount of bombs dropped at a time by 1;
   2. Flames: Increases explosion range from the bomb in four directions by 1 block;
   3. Speed: Increases movement speed;

When the user opens the game, he/she should be presented to a page where he/she should enter a **nickname** to differentiate users. After selecting a nickname the user should be presented to a waiting page with a **player counter** that ends at 4. Once a user joins, the player counter will increment by 1.

If there are more than 2 players in the counter and it does not reach 4 players before 20 seconds, a 10 second timer starts, to players get ready to start the game.\
If there are 4 players in the counter before 20 seconds, the 10 seconds timer starts and the game starts.

#### Bonus

Although this bomberman already is super cool, it can be always better. Here are some ideas you can implement into the game to make it super awesomely cool:

- **Solo + Co-Op mode**: You are supposed to develop an AI to play against the players. So once the AI is defeated all players involved win.
- **More power ups**:

  - _Bomb Pass_: Ability to pass through bombs;
  - _Block Pass_: Ability to pass through blocks (not walls);
  - _Detonator_: Ability to choose when a bomb will explode on a key press;
  - _1 Up_: Gives the player an extra life;
- **Release power ups after death**: When a player dies it drops one of it's power ups. If the player had no power ups, it drops a random power up.
- **Team mode**: Make games with 2v2 (two players versus two players).
- **After death interaction**: When a player dies, they can reappear as a ghost. If a ghost touches another player they come back to life. If a ghost is caught in a bomb explosion, the player controlling the ghost dies permanently.

// M0 static starter. Yacobe adds the WebSocket transport and room here in M1.
import { createServer } from "node:http";
import { readFile, realpath, stat } from "node:fs/promises";
import { dirname, extname, isAbsolute, relative, resolve, sep } from "node:path";
import { fileURLToPath, pathToFileURL } from "node:url";

const projectRoot = resolve(dirname(fileURLToPath(import.meta.url)), "..");
const publicRoot = resolve(projectRoot, "public");
const sharedRoot = resolve(projectRoot, "shared");
const mimeTypes = {
  ".html": "text/html; charset=utf-8",
  ".js": "text/javascript; charset=utf-8",
  ".css": "text/css; charset=utf-8",
  ".json": "application/json; charset=utf-8",
  ".png": "image/png", ".jpg": "image/jpeg", ".jpeg": "image/jpeg",
  ".ico": "image/x-icon", ".woff2": "font/woff2",
};

function inside(root, path) {
  const suffix = relative(root, path);
  return suffix !== ".." && !suffix.startsWith(`..${sep}`) && !isAbsolute(suffix);
}

export function createAppServer() {
  return createServer(async (request, response) => {
    response.setHeader("Cache-Control", "no-store");
    response.setHeader("X-Content-Type-Options", "nosniff");
    if (request.method !== "GET" && request.method !== "HEAD") {
      response.writeHead(405, { Allow: "GET, HEAD" }).end();
      return;
    }
    try {
      const pathname = decodeURIComponent(request.url.split("?")[0]);
      const isShared = pathname.startsWith("/shared/");
      const root = isShared ? sharedRoot : publicRoot;
      const route = isShared ? pathname.slice("/shared/".length) : pathname.slice(1);
      const candidate = resolve(root, route || "index.html");
      if (!inside(root, candidate)) {
        response.writeHead(403).end();
        return;
      }
      const path = await realpath(candidate);
      if (!inside(root, path)) {
        response.writeHead(403).end();
        return;
      }
      if (!(await stat(path)).isFile()) {
        response.writeHead(404).end();
        return;
      }
      const body = await readFile(path);
      response.writeHead(200, {
        "Content-Type": mimeTypes[extname(path)] ?? "application/octet-stream",
        "Content-Length": body.length,
      });
      response.end(request.method === "HEAD" ? undefined : body);
    } catch (error) {
      const status = error instanceof URIError ? 400
        : ["ENOENT", "ENOTDIR", "EISDIR"].includes(error.code) ? 404 : 500;
      response.writeHead(status).end();
      if (status === 500) console.error("Static request failed:", error);
    }
  });
}

if (process.argv[1] && import.meta.url === pathToFileURL(resolve(process.argv[1])).href) {
  const server = createAppServer();
  server.on("error", error => {
    console.error("Server failed:", error.message);
    process.exitCode = 1;
  });
  server.listen(Number(process.env.PORT ?? 8080), () => {
    console.log(`Bomberman starter: http://localhost:${server.address().port}`);
  });
}

# Runtime mini-framework

This is the canonical framework used by Bomberman. Hichame owns maintenance.
This is the only framework copy in the repository. Make all framework changes
here. No build step or client dependencies.

Import public APIs from `index.js`: `h`, `render`, `State`, event helpers,
`listen`, and `createRouter`.

## DOM and events

`h(tag, props, ...children)` builds a vnode; `render(vnode, element)` mounts or
patches it. Children are flattened, numbers become text, and null/booleans are
omitted. User content is text. Children reconcile by position, not key. Keep
the board grid and four player slots structurally stable.

**Rendered vnodes, props, style objects and children are immutable.** Reusing
the same vnode at the same position skips its entire subtree. Build a fresh
vnode to change it. Do not modify framework-managed DOM externally, except the
agreed input/focus/scroll operations. Equal virtual text is not expected to
repair manually tampered DOM.

`style` accepts a CSS string or a plain object. Object keys may be camelCase,
hyphenated properties, or custom properties. Include units explicitly:

```js
h("div", { style: { transform: "translate3d(20px, 0, 0)", "--tile": "40px" } });
```

Removed style keys are cleared. String/object/removal transitions work.
Event props (`onClick`, `onInput`, `onChange`, `onSubmit`, `onKeyDown`, `onKeyUp`,
`onDblClick`, `onFocus`, `onBlur`) receive native events. Use stable handlers in
per-frame trees. Removed subtrees have their listeners cleaned recursively.

`listen(target, type, handler, options)` registers global events and returns a
cleanup function. Use it for keyboard/blur and focusin/visibilitychange events;
call cleanup on disposal.

## State

`new State(initialState)` preserves synchronous notification timing. Bomberman
opts into batching: `new State(initialState, { batch: "microtask" })`.
`batch: "raf"` is available in a browser; do not use it for positions or create
a second game-rendering loop around it.

`setState(objectOrUpdater)` shallow-merges immediately; functional updaters see
the latest state even within a batch. Null, non-object, array, empty and unchanged
top-level patches do not notify. Update nested data immutably. Listeners receive
`(state, version)`; each accepted patch increments version. Listener errors are
logged without preventing the remaining listeners from running.
`subscribe(listener)` returns an unsubscribe function.

From the project root, run `npm test`, then `npm start` and open
`/dev/framework-check.html` for real-browser checks. TodoMVC source was not
included in the supplied snapshot; its original end-to-end regression check
requires that source separately.

// Framework Client-Side Router (Hash-based)

export function createRouter(routes = {}) {
  // Get current route path from URL hash (e.g. "#/active" -> "/active")
  function getPath() {
    const hash = window.location.hash.slice(1); // remove '#'
    return hash || '/'; // Default route if hash is empty
  }

  // Find matching route and run its callback function
  function handleRoute() {
    const currentPath = getPath();
    const routeCallback = routes[currentPath] || routes['/']; // Fallback to '/' if route not found
    
    if (typeof routeCallback === 'function') {
      routeCallback(currentPath);
    }
  }

  return {
    // Navigate programmatically to a new path (e.g., router.navigate('/active'))
    navigate(path) {
      window.location.hash = path;
    },

    // Start listening for URL hash changes and execute initial route
    start() {
      // Listen to browser hash changes (Back/Forward buttons or clicking links)
      window.addEventListener('hashchange', handleRoute);

      // Execute route callback immediately on page load
      handleRoute();
    }
  };
}

function updateProps(element, oldProps = {}, newProps = {}) {
  const keys = new Set([...Object.keys(oldProps), ...Object.keys(newProps)]);

  for (const key of keys) {
    const oldValue = oldProps[key];
    const newValue = newProps[key];

    if (oldValue === newValue) continue;
    if (key === "key") continue;

    if (key === "style") {
      applyStyle(element, oldValue, newValue);
      continue;
    }

    if (key.startsWith("on")) {
      addEvent(element, key, newValue);
      continue;
    }

    if (key === "value" && "value" in element) {
      const value = newValue == null ? "" : String(newValue);

      if (element.value !== value) {
        element.value = value;
      }

      continue;
    }

    if (key === "checked" && "checked" in element) {
      element.checked = Boolean(newValue);
      continue;
    }

    if (key === "selected" && "selected" in element) {
      element.selected = Boolean(newValue);
      continue;
    }
    if (newValue == null || newValue === false) {
      element.removeAttribute(key);
      continue;
    }

    if (newValue === true) {
      element.setAttribute(key, "");
      continue;
    }

    element.setAttribute(key, String(newValue));
  }
}

// Reconcile a list of children positionally inside `parent`.
function patchChildren(parent, oldChildren = [], newChildren = []) {
  const sharedLength = Math.min(oldChildren.length, newChildren.length);

  for (let i = 0; i < sharedLength; i++) {
    updateDOM(parent, newChildren[i], oldChildren[i], i);
  }

  for (let i = oldChildren.length - 1; i >= newChildren.length; i--) {
    const childToRemove = parent.childNodes[i];
    if (childToRemove) {
      removeEvents(childToRemove);
      parent.removeChild(childToRemove);
    }
  }

  for (let i = oldChildren.length; i < newChildren.length; i++) {
    parent.appendChild(createDom(newChildren[i]));
  }
}

function updateDOM(parent, newVNode, oldVNode, index = 0) {
  // Rendered vnodes are immutable. Reusing one skips the entire subtree.
  if (newVNode === oldVNode) return;

  const current = parent.childNodes[index];

  if (newVNode == null) {
    if (current) {
      removeEvents(current); 
      parent.removeChild(current);
    }
    return;
  }

  if (oldVNode == null) {
    parent.insertBefore(createDom(newVNode), current ?? null);
    return;
  }

  const newIsPrimitive =
    typeof newVNode === "string" || typeof newVNode === "number";

  const oldIsPrimitive =
    typeof oldVNode === "string" || typeof oldVNode === "number";

  if (newIsPrimitive || oldIsPrimitive) {
    if (
      newIsPrimitive &&
      oldIsPrimitive &&
      typeof newVNode === typeof oldVNode
    ) {
      if (String(newVNode) !== String(oldVNode)) {
        current.nodeValue = String(newVNode);
      }
    } else {
      removeEvents(current); 
      parent.replaceChild(createDom(newVNode), current);
    }

    return;
  }

  if (newVNode.type !== oldVNode.type) {
    removeEvents(current); 
    parent.replaceChild(createDom(newVNode), current);
    return;
  }

  updateProps(current, oldVNode.props, newVNode.props);

  patchChildren(current, oldVNode.children ?? [], newVNode.children ?? []);
}

export default function patch(oldVNode, newVNode, container) {
  if (oldVNode === newVNode) return;

  // A fragment root is a flat list of siblings inside the container.
  if (Array.isArray(oldVNode) || Array.isArray(newVNode)) {
    patchChildren(
      container,
      Array.isArray(oldVNode) ? oldVNode : [oldVNode],
      Array.isArray(newVNode) ? newVNode : [newVNode]
    );
    return;
  }

  updateDOM(container, newVNode, oldVNode, 0);
}

export default function render(vnode, container) {

    throw new TypeError("render(): container must be a valid DOM element");
  }

  if (!vnode || typeof vnode !== "object") {
    throw new TypeError("render(): vnode must be a valid VNode");
  }

  const oldVNode = previousVNodes.get(container);

  if (!oldVNode) {
    const dom = createDom(vnode);
    container.replaceChildren(dom);
  } else {
    patch(oldVNode, vnode, container);
  }

  previousVNodes.set(container, vnode);
}

export default function createDom(vnode) {
  // A fragment (array of vnodes) renders as sibling nodes with no wrapper element.
  if (Array.isArray(vnode)) {
    const fragment = document.createDocumentFragment();

    vnode.forEach((child) => {
      fragment.appendChild(createDom(child));
    });

    return fragment;
  }

  if (typeof vnode === "string" || typeof vnode === "number") {
    return document.createTextNode(vnode);
  }

  let element = document.createElement(vnode.type);

  if (vnode.props) {
     addprop(vnode, element);
  }

  if (vnode.children) {
    vnode.children.forEach((child) => {
      element.appendChild(createDom(child));
    });
  }

  return element;
}

function clearStyle(element) {
  // Synchronize pending CSSOM writes before removing the attribute.
  if (element.hasAttribute("style")) element.removeAttribute("style");
}

function isStyleObject(value) {
  if (value === null || typeof value !== "object") return false;
  const prototype = Object.getPrototypeOf(value);
  return prototype === Object.prototype || prototype === null;
}

function setStyleProp(element, name, value) {
  const text = value == null || value === false ? "" : String(value);
  if (name.includes("-")) {
    if (text === "") element.style.removeProperty(name);
    else element.style.setProperty(name, text);
  } else {
    element.style[name] = text;
  }
}

export default function applyStyle(element, oldStyle, newStyle) {
  if (newStyle == null || newStyle === false || newStyle === "") {
    clearStyle(element);
    return;
  }
  if (typeof newStyle === "string") {
    element.setAttribute("style", newStyle);
    return;
  }
  if (!isStyleObject(newStyle)) {
    throw new TypeError("style must be a CSS string or a plain object");
  }
  if (!isStyleObject(oldStyle)) {
    clearStyle(element);
    for (const [name, value] of Object.entries(newStyle)) setStyleProp(element, name, value);
    return;
  }
  for (const name of Object.keys(oldStyle)) {
    if (!Object.hasOwn(newStyle, name)) setStyleProp(element, name, "");
  }
  for (const [name, value] of Object.entries(newStyle)) {
    if (oldStyle[name] !== value) setStyleProp(element, name, value);
  }
}

const BATCH_MODES = ["sync", "microtask", "raf"];

export class State {
  constructor(initialState = {}, { batch = "sync" } = {}) {
    if (!BATCH_MODES.includes(batch)) {
      throw new TypeError(`State: batch must be one of ${BATCH_MODES.join(", ")}`);
    }
    this.state = initialState;
    this.listeners = [];
    this.version = 0;
    this.batch = batch;
    this.pending = false;
  }

  getState() {
    return this.state;
  }

  setState(update) {
    const changes = typeof update === "function" ? update(this.state) : update;
    if (changes == null || typeof changes !== "object" || Array.isArray(changes)) return;
    const keys = Object.keys(changes);
    if (keys.length === 0 || keys.every(key => Object.is(this.state[key], changes[key]))) return;

    this.state = { ...this.state, ...changes };
    this.version++;
    this.schedule();
  }

  schedule() {
    if (this.batch === "sync") {
      this.notify();
      return;
    }
    if (this.pending) return;
    this.pending = true;
    const run = () => {
      this.pending = false;
      this.notify();
    };
    if (this.batch === "raf") requestAnimationFrame(run);
    else queueMicrotask(run);
  }

  notify() {
    const state = this.state;
    const version = this.version;
    for (const listener of [...this.listeners]) {
      try {
        listener(state, version);
      } catch (error) {
        console.error("State listener failed:", error);
      }
    }
  }

  subscribe(listener) {
    if (typeof listener !== "function") {
      throw new TypeError("State.subscribe(): listener must be a function");
    }
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(item => item !== listener);
    };
  }
}

//EVENT_MAP (Translate) -> addEvent (Clean Old + Attach New) -> handleKey (Shortcuts) -> removeEvents (RAM Cleanup)
// Map framework props to native browser DOM events
const EVENT_MAP = {
  onClick: 'click',
  onInput: 'input',
  onChange: 'change',
  onKeyDown: 'keydown',
  onKeyUp: 'keyup',
  onSubmit: 'submit',
  onDblClick: 'dblclick',
  onBlur: 'blur',
  onFocus: 'focus'
};

// Remove one custom event listener from an element
export function removeEvent(element, propName) {
  const nativeEvent = EVENT_MAP[propName];

  if (!nativeEvent || !element._listeners) return;

  const handler = element._listeners[propName];

  if (!handler) return;

  element.removeEventListener(nativeEvent, handler);
  delete element._listeners[propName];

  if (Object.keys(element._listeners).length === 0) {
    delete element._listeners;
  }
}

// Attach custom event listener and prevent duplicate listeners
export function addEvent(element, propName, handler) {
  // Get native DOM event name (e.g., 'onClick' -> 'click')
  const nativeEvent = EVENT_MAP[propName];

  // Stop if the event name is unsupported
  if (!nativeEvent) return;

  // A missing/non-function handler means the prop was removed.
  if (typeof handler !== 'function') {
    removeEvent(element, propName);
    return;
  }

  // Create listener storage object if it does not exist
  if (!element._listeners) {
    element._listeners = {};
  }

  // Remove old event listener if it already exists
  if (element._listeners[propName]) {
    element.removeEventListener(nativeEvent, element._listeners[propName]);
  }

  // Save new handler in element storage
  element._listeners[propName] = handler;

  // Add new event listener to the native DOM element
  element.addEventListener(nativeEvent, handler);
}

// Remove all event listeners when element is deleted from DOM
export function removeEvents(element) {
  if (!element) return;

  // Loop through all saved event handlers
  for (const propName of Object.keys(element._listeners ?? {})) {
    const nativeEvent = EVENT_MAP[propName];
    if (nativeEvent) {
      // Remove each event listener from browser RAM
      element.removeEventListener(nativeEvent, element._listeners[propName]);
    }
  }

  // Delete storage property to completely clear memory
  delete element._listeners;
  for (const child of element.children ?? []) removeEvents(child);
}

export function listen(target, type, handler, options) {
  target.addEventListener(type, handler, options);
  return () => target.removeEventListener(type, handler, options);
}

// Helper function for handling keyboard shortcuts (Enter & Escape)
export function handleKey(e, onEnter, onEscape) {
  // Execute onEnter callback if Enter key is pressed
  if (e.key === 'Enter' && onEnter) {
    onEnter(e);
  // Execute onEscape callback if Escape key is pressed
  } else if (e.key === 'Escape' && onEscape) {
    onEscape(e);
  }
}
